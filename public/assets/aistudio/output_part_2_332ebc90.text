package com.example.data

import android.content.Context
import android.net.Uri
import android.provider.DocumentsContract
import android.provider.OpenableColumns
import com.example.model.FirmwareInfoSummary
import com.example.model.FlashFileType
import com.example.model.FlashStepStatus
import com.example.model.ParsedFlashStep
import com.example.model.SelectedFlashFile
import org.xmlpull.v1.XmlPullParser
import org.xmlpull.v1.XmlPullParserFactory

import java.nio.ByteOrder

object FlashFileProcessor {

    private const val LARGE_PARTITION_THRESHOLD_BYTES = SparseChunkPlanner.DEFAULT_SAFE_CHUNK_BYTES

    data class FolderDocEntry(
        val relativePath: String,
        val fileName: String,
        val uri: Uri,
        val sizeBytes: Long
    )

    fun inspectAndParseUri(context: Context, uri: Uri, defaultPartition: String): SelectedFlashFile {
        val resolver = context.contentResolver
        var displayName = uri.lastPathSegment?.substringAfterLast('/') ?: "firmware.bin"
        var fileSize = -1L

        try {
            resolver.query(uri, null, null, null, null)?.use { cursor ->
                val nameIdx = cursor.getColumnIndex(OpenableColumns.DISPLAY_NAME)
                val sizeIdx = cursor.getColumnIndex(OpenableColumns.SIZE)
                if (cursor.moveToFirst()) {
                    if (nameIdx >= 0) {
                        cursor.getString(nameIdx)?.takeIf { it.isNotBlank() }?.let { displayName = it }
                    }
                    if (sizeIdx >= 0 && !cursor.isNull(sizeIdx)) {
                        fileSize = cursor.getLong(sizeIdx)
                    }
                }
            }
        } catch (_: Exception) {
        }

        if (fileSize <= 0L) {
            try {
                resolver.openFileDescriptor(uri, "r")?.use { pfd ->
                    if (pfd.statSize > 0L) {
                        fileSize = pfd.statSize
                    }
                }
            } catch (_: Exception) {
            }
        }

        val header = readHeaderBytes(context, uri, 28)
        val headerRead = header.size
        val magicHex = if (headerRead > 0) {
            header.take(headerRead.coerceAtMost(8))
                .joinToString(" ") { String.format(Locale.US, "%02X", it) }
        } else {
            "EMPTY"
        }

        val sparseDetails = parseSparseHeaderDescription(header)
        val isSparse = sparseDetails != null
        val hasHuaweiMagic = checkUpdateAppMagic(context, uri)
        val lower = displayName.lowercase(Locale.US)

        val probeText = if (lower.endsWith(".txt") || lower.endsWith(".xml") ||
            lower.endsWith(".bat") || lower.endsWith(".cmd") || lower.endsWith(".sh")
        ) {
            readTextSafely(context, uri, maxBytes = 2 * 1024 * 1024)
        } else {
            ""
        }

        val type = when {
            lower.endsWith(".app") || hasHuaweiMagic -> FlashFileType.UPDATE_APP
            lower.contains("scatter") && lower.endsWith(".txt") ||
                probeText.contains("partition_index", ignoreCase = true) -> FlashFileType.SCATTER
            lower.endsWith(".xml") &&
                (lower.contains("rawprogram") || probeText.contains("<program", ignoreCase = true)) ->
                FlashFileType.RAWPROGRAM
            lower.endsWith(".bat") || lower.endsWith(".cmd") || lower.endsWith(".sh") ->
                FlashFileType.SCRIPT
            lower.endsWith(".zip") -> FlashFileType.ARCHIVE
            lower.endsWith(".img") || lower.endsWith(".bin") || isSparse -> FlashFileType.IMAGE
            else -> FlashFileType.RAW
        }

        return when (type) {
            FlashFileType.SCATTER -> {
                val (steps, fwInfo) = parseScatterContent(
                    text = probeText,
                    folderFiles = emptyMap(),
                    sparseChecker = { false }
                )
                SelectedFlashFile(
                    uriString = uri.toString(),
                    fileName = displayName,
                    fileSizeBytes = fileSize.coerceAtLeast(0L),
                    formattedSize = formatSize(fileSize),
                    fileType = type,
                    magicHeaderHex = magicHex,
                    isSparseImage = false,
                    sparseInfo = "MTK Scatter Partition Map",
                    steps = steps,
                    firmwareInfo = fwInfo
                )
            }

            FlashFileType.RAWPROGRAM -> {
                val (steps, fwInfo) = parseRawprogramXmlContent(
                    xmlList = listOf(displayName to probeText),
                    folderFiles = emptyMap(),
                    sparseChecker = { false }
                )
                SelectedFlashFile(
                    uriString = uri.toString(),
                    fileName = displayName,
                    fileSizeBytes = fileSize.coerceAtLeast(0L),
                    formattedSize = formatSize(fileSize),
                    fileType = type,
                    magicHeaderHex = magicHex,
                    isSparseImage = false,
                    sparseInfo = "Qualcomm Firehose Partition Table",
                    steps = steps,
                    firmwareInfo = fwInfo
                )
            }

            FlashFileType.SCRIPT -> {
                val (steps, fwInfo) = parseScriptContent(
                    scriptText = probeText,
                    fileName = displayName,
                    folderFiles = emptyMap(),
                    sparseChecker = { false }
                )
                SelectedFlashFile(
                    uriString = uri.toString(),
                    fileName = displayName,
                    fileSizeBytes = fileSize.coerceAtLeast(0L),
                    formattedSize = formatSize(fileSize),
                    fileType = type,
                    magicHeaderHex = magicHex,
                    isSparseImage = false,
                    sparseInfo = "Fastboot Flash Script",
                    steps = steps,
                    firmwareInfo = fwInfo
                )
            }

            FlashFileType.UPDATE_APP -> {
                val (steps, fwInfo) = parseRealUpdateAppWithChannel(context, uri)
                SelectedFlashFile(
                    uriString = uri.toString(),
                    fileName = displayName,
                    fileSizeBytes = fileSize.coerceAtLeast(0L),
                    formattedSize = formatSize(fileSize),
                    fileType = type,
                    magicHeaderHex = magicHex,
                    isSparseImage = false,
                    sparseInfo = "Huawei UPDATE.APP Container",
                    steps = steps,
                    firmwareInfo = fwInfo
                )
            }

            FlashFileType.ARCHIVE -> {
                val steps = parseRealZipEntries(context, uri)
                val totalPayload = steps.sumOf { it.dataSize }
                val largeParts = steps.filter { it.dataSize > LARGE_PARTITION_THRESHOLD_BYTES }
                    .map { it.targetPartition }
                SelectedFlashFile(
                    uriString = uri.toString(),
                    fileName = displayName,
                    fileSizeBytes = fileSize.coerceAtLeast(0L),
                    formattedSize = formatSize(fileSize),
                    fileType = type,
                    magicHeaderHex = magicHex,
                    isSparseImage = false,
                    sparseInfo = "ZIP Firmware Archive",
                    steps = steps,
                    firmwareInfo = FirmwareInfoSummary(
                        formatLabel = "ZIP Firmware Package",
                        chipsetOrPlatform = "Universal / OTA",
                        projectOrBoard = displayName.substringBeforeLast('.'),
                        storageType = "Archive",
                        resolvedImageCount = steps.size,
                        missingImageCount = 0,
                        largePartitionNames = largeParts,
                        extraMetadataLines = listOf("Uncompressed Payload: ${formatSize(totalPayload)}")
                    )
                )
            }

            FlashFileType.IMAGE, FlashFileType.RAW -> {
                val inferred = inferPartition(lower, defaultPartition)
                val safeSize = fileSize.coerceAtLeast(0L)
                val largeParts = if (safeSize > LARGE_PARTITION_THRESHOLD_BYTES) listOf(inferred) else emptyList()
                val step = ParsedFlashStep(
                    id = "img_1",
                    order = 1,
                    command = "fastboot flash $inferred $displayName",
                    targetPartition = inferred,
                    sourceName = displayName,
                    sourceUriString = uri.toString(),
                    dataOffset = 0L,
                    dataSize = safeSize,
                    isSparse = isSparse,
                    isFlashable = true,
                    status = FlashStepStatus.READY
                )
                SelectedFlashFile(
                    uriString = uri.toString(),
                    fileName = displayName,
                    fileSizeBytes = safeSize,
                    formattedSize = formatSize(safeSize),
                    fileType = type,
                    magicHeaderHex = magicHex,
                    isSparseImage = isSparse,
                    sparseInfo = sparseDetails ?: "Raw Binary Stream",
                    steps = listOf(step),
                    firmwareInfo = FirmwareInfoSummary(
                        formatLabel = if (isSparse) "Android Sparse Image" else "Raw Partition Image",
                        chipsetOrPlatform = "Single Partition Target: $inferred",
                        projectOrBoard = displayName,
                        storageType = if (isSparse) "Sparse (0xED26FF3A)" else "Raw Ext4/Binary",
                        resolvedImageCount = 1,
                        missingImageCount = 0,
                        largePartitionNames = largeParts,
                        extraMetadataLines = listOf(sparseDetails ?: "Direct Stream (${formatSize(safeSize)})")
                    )
                )
            }
        }
    }

    /**
     * Walks a SAF DocumentTree folder (root + images/ subfolders) to discover Scatter,
     * Rawprogram XML, Flash .bat/.sh, or UPDATE.APP packages and resolves all partition image URIs.
     */
    fun inspectAndParseFolderTree(
        context: Context,
        treeUri: Uri,
        defaultPartition: String
    ): SelectedFlashFile? {
        val entries = collectFolderEntries(context, treeUri)
        if (entries.isEmpty()) return null

        val folderName = runCatching {
            DocumentsContract.getTreeDocumentId(treeUri).substringAfterLast('/').substringAfterLast(':')
        }.getOrNull()?.takeIf { it.isNotBlank() } ?: "Firmware Folder"

        val byRelative = entries.associateBy { it.relativePath.lowercase(Locale.US) }
        val byName = mutableMapOf<String, FolderDocEntry>()
        entries.forEach { entry ->
            val key = entry.fileName.lowercase(Locale.US)
            if (!byName.containsKey(key)) {
                byName[key] = entry
            }
        }

        val scatterDoc = entries.firstOrNull {
            val l = it.fileName.lowercase(Locale.US)
            l.endsWith(".txt") && l.contains("scatter")
        }
        val rawprogramDocs = entries.filter {
            val l = it.fileName.lowercase(Locale.US)
            l.endsWith(".xml") && l.contains("rawprogram")
        }.sortedBy { it.fileName.lowercase(Locale.US) }

        val scriptDoc = entries.filter {
            val l = it.fileName.lowercase(Locale.US)
            l.endsWith(".bat") || l.endsWith(".cmd") || l.endsWith(".sh")
        }.sortedWith(
            compareBy<FolderDocEntry> {
                val l = it.fileName.lowercase(Locale.US)
                when {
                    l.startsWith("flash_all_except") -> 0
                    l.startsWith("flash_all") -> 1
                    l.contains("flash") -> 2
                    else -> 3
                }
            }.thenBy { it.fileName }
        ).firstOrNull()

        val updateAppDoc = entries.firstOrNull {
            it.fileName.equals("UPDATE.APP", ignoreCase = true) ||
                it.fileName.lowercase(Locale.US).endsWith(".app")
        }

        val sparseCheck: (Uri) -> Boolean = { imgUri ->
            val hdr = readHeaderBytes(context, imgUri, 4)
            isSparseMagic(hdr)
        }

        val allLookupMap = HashMap<String, FolderDocEntry>(byRelative.size + byName.size).apply {
            putAll(byName)
            putAll(byRelative)
        }

        return when {
            scatterDoc != null -> {
                val text = readTextSafely(context, scatterDoc.uri, maxBytes = 2 * 1024 * 1024)
                val (steps, fwInfo) = parseScatterContent(text, allLookupMap, sparseCheck)
                val totalImgBytes = steps.filter { it.sourceUriString != null }.sumOf { it.dataSize }
                SelectedFlashFile(
                    uriString = scatterDoc.uri.toString(),
                    fileName = "$folderName/${scatterDoc.fileName}",
                    fileSizeBytes = totalImgBytes,
                    formattedSize = formatSize(totalImgBytes),
                    fileType = FlashFileType.SCATTER,
                    magicHeaderHex = "SCATTER",
                    isSparseImage = steps.any { it.isSparse },
                    sparseInfo = "MTK Scatter Package",
                    steps = steps,
                    firmwareInfo = fwInfo.copy(isFolderPackage = true)
                )
            }

            rawprogramDocs.isNotEmpty() -> {
                val xmlPairs = rawprogramDocs.map { doc ->
                    doc.fileName to readTextSafely(context, doc.uri, maxBytes = 2 * 1024 * 1024)
                }
                val (steps, fwInfo) = parseRawprogramXmlContent(xmlPairs, allLookupMap, sparseCheck)
                val totalImgBytes = steps.filter { it.sourceUriString != null }.sumOf { it.dataSize }
                SelectedFlashFile(
                    uriString = rawprogramDocs.first().uri.toString(),
                    fileName = "$folderName (${rawprogramDocs.size} XML)",
                    fileSizeBytes = totalImgBytes,
                    formattedSize = formatSize(totalImgBytes),
                    fileType = FlashFileType.RAWPROGRAM,
                    magicHeaderHex = "XML",
                    isSparseImage = steps.any { it.isSparse },
                    sparseInfo = "Qualcomm Firehose Package",
                    steps = steps,
                    firmwareInfo = fwInfo.copy(isFolderPackage = true)
                )
            }

            scriptDoc != null -> {
                val text = readTextSafely(context, scriptDoc.uri, maxBytes = 1024 * 1024)
                val (steps, fwInfo) = parseScriptContent(text, scriptDoc.fileName, allLookupMap, sparseCheck)
                val totalImgBytes = steps.filter { it.sourceUriString != null }.sumOf { it.dataSize }
                SelectedFlashFile(
                    uriString = scriptDoc.uri.toString(),
                    fileName = "$folderName/${scriptDoc.fileName}",
                    fileSizeBytes = totalImgBytes,
                    formattedSize = formatSize(totalImgBytes),
                    fileType = FlashFileType.SCRIPT,
                    magicHeaderHex = "SCRIPT",
                    isSparseImage = steps.any { it.isSparse },
                    sparseInfo = "Fastboot Script Package",
                    steps = steps,
                    firmwareInfo = fwInfo.copy(isFolderPackage = true)
                )
            }

            updateAppDoc != null -> {
                inspectAndParseUri(context, updateAppDoc.uri, defaultPartition)
            }

            else -> {
                val imageEntries = entries.filter {
                    val l = it.fileName.lowercase(Locale.US)
                    l.endsWith(".img") || l.endsWith(".bin")
                }.sortedBy { it.fileName }
                if (imageEntries.isEmpty()) return null

                val steps = imageEntries.mapIndexed { idx, entry ->
                    val part = inferPartition(entry.fileName.lowercase(Locale.US), defaultPartition)
                    val isSp = sparseCheck(entry.uri)
                    ParsedFlashStep(
                        id = "fimg_${idx + 1}",
                        order = idx + 1,
                        command = "fastboot flash $part ${entry.fileName}",
                        targetPartition = part,
                        sourceName = entry.relativePath,
                        sourceUriString = entry.uri.toString(),
                        dataOffset = 0L,
                        dataSize = entry.sizeBytes,
                        isSparse = isSp,
                        isFlashable = true,
                        status = FlashStepStatus.READY
                    )
                }
                val totalBytes = steps.sumOf { it.dataSize }
                val largeParts = steps.filter { it.dataSize > LARGE_PARTITION_THRESHOLD_BYTES }
                    .map { it.targetPartition }
                SelectedFlashFile(
                    uriString = treeUri.toString(),
                    fileName = "$folderName (${steps.size} images)",
                    fileSizeBytes = totalBytes,
                    formattedSize = formatSize(totalBytes),
                    fileType = FlashFileType.IMAGE,
                    magicHeaderHex = "FOLDER",
                    isSparseImage = steps.any { it.isSparse },
                    sparseInfo = "Multi-Image Folder",
                    steps = steps,
                    firmwareInfo = FirmwareInfoSummary(
                        formatLabel = "Fastboot Image Folder",
                        chipsetOrPlatform = "Universal Fastboot",
                        projectOrBoard = folderName,
                        storageType = "Multi-Partition",
                        isFolderPackage = true,
                        resolvedImageCount = steps.size,
                        missingImageCount = 0,
                        largePartitionNames = largeParts
                    )
                )
            }
        }
    }

    private fun collectFolderEntries(
        context: Context,
        treeUri: Uri
    ): List<FolderDocEntry> {
        val result = mutableListOf<FolderDocEntry>()
        val rootDocId = runCatching { DocumentsContract.getTreeDocumentId(treeUri) }.getOrNull() ?: return emptyList()

        fun scanDir(docId: String, prefix: String, depth: Int) {
            if (depth > 2) return
            val childrenUri = DocumentsContract.buildChildDocumentsUriUsingTree(treeUri, docId)
            val projection = arrayOf(
                DocumentsContract.Document.COLUMN_DOCUMENT_ID,
                DocumentsContract.Document.COLUMN_DISPLAY_NAME,
                DocumentsContract.Document.COLUMN_MIME_TYPE,
                DocumentsContract.Document.COLUMN_SIZE
            )
            try {
                context.contentResolver.query(childrenUri, projection, null, null, null)?.use { c ->
                    val idIdx = c.getColumnIndex(DocumentsContract.Document.COLUMN_DOCUMENT_ID)
                    val nameIdx = c.getColumnIndex(DocumentsContract.Document.COLUMN_DISPLAY_NAME)
                    val mimeIdx = c.getColumnIndex(DocumentsContract.Document.COLUMN_MIME_TYPE)
                    val sizeIdx = c.getColumnIndex(DocumentsContract.Document.COLUMN_SIZE)

                    while (c.moveToNext()) {
                        val childId = if (idIdx >= 0) c.getString(idIdx) else continue
                        val name = if (nameIdx >= 0) c.getString(nameIdx) else continue
                        val mime = if (mimeIdx >= 0) c.getString(mimeIdx) else ""
                        val size = if (sizeIdx >= 0 && !c.isNull(sizeIdx)) c.getLong(sizeIdx) else 0L
                        val relPath = if (prefix.isEmpty()) name else "$prefix/$name"

                        if (mime == DocumentsContract.Document.MIME_TYPE_DIR) {
                            scanDir(childId, relPath, depth + 1)
                        } else {
                            val fileUri = DocumentsContract.buildDocumentUriUsingTree(treeUri, childId)
                            result.add(
                                FolderDocEntry(
                                    relativePath = relPath,
                                    fileName = name,
                                    uri = fileUri,
                                    sizeBytes = size.coerceAtLeast(0L)
                                )
                            )
                        }
                    }
                }
            } catch (_: Exception) {
            }
        }

        scanDir(rootDocId, "", 0)
        return result
    }

    /**
     * Parses MediaTek Android Scatter files (both classic INI and YAML `- partition_index:` styles).
     */
    fun parseScatterContent(
        text: String,
        folderFiles: Map<String, FolderDocEntry>,
        sparseChecker: (Uri) -> Boolean
    ): Pair<List<ParsedFlashStep>, FirmwareInfoSummary> {
        var configVersion = ""
        var platform = "MediaTek"
        var project = "Unknown"
        var storage = "EMMC"
        var bootChannel = ""

        data class MutableBlock(
            var index: String = "",
            var partName: String = "",
            var fileName: String = "",
            var isDownload: Boolean = false,
            var startAddr: String = "",
            var partitionSize: Long = 0L,
            var type: String = ""
        )

        val blocks = mutableListOf<MutableBlock>()
        var current: MutableBlock? = null
        var inGeneralSection = true

        text.lineSequence().forEach { raw ->
            val trimmed = raw.trim()
            if (trimmed.isEmpty() || trimmed.startsWith("#")) return@forEach

            val clean = trimmed.removePrefix("-").trim()
            val colonIdx = clean.indexOf(':')
            if (colonIdx <= 0) return@forEach

            val key = clean.substring(0, colonIdx).trim().lowercase(Locale.US)
            val value = clean.substring(colonIdx + 1).trim().trim('"', '\'')

            if (key == "partition_index") {
                inGeneralSection = false
                current = MutableBlock(index = value).also { blocks.add(it) }
                return@forEach
            }

            if (inGeneralSection) {
                when (key) {
                    "config_version" -> configVersion = value
                    "platform" -> platform = value
                    "project" -> project = value
                    "storage" -> storage = value
                    "boot_channel" -> bootChannel = value
                }
            }

            val blk = current ?: return@forEach
            when (key) {
                "partition_name" -> blk.partName = value
                "file_name" -> blk.fileName = value
                "is_download" -> blk.isDownload = value.equals("true", ignoreCase = true)
                "linear_start_addr", "physical_start_addr" -> {
                    if (blk.startAddr.isEmpty() || key == "linear_start_addr") {
                        blk.startAddr = value
                    }
                }
                "partition_size" -> blk.partitionSize = parseHexOrDecLong(value)
                "type" -> blk.type = value
            }
        }

        var resolvedCount = 0
        var missingCount = 0
        val steps = mutableListOf<ParsedFlashStep>()

        blocks.forEachIndexed { idx, blk ->
            if (blk.partName.isBlank()) return@forEachIndexed
            val cleanPart = blk.partName.lowercase(Locale.US)
            val hasFile = blk.fileName.isNotBlank() && !blk.fileName.equals("NONE", ignoreCase = true)
            val docEntry = if (hasFile) resolveFolderFile(blk.fileName, folderFiles) else null
            val actualSize = docEntry?.sizeBytes?.takeIf { it > 0L } ?: blk.partitionSize
            val isSparse = docEntry != null && sparseChecker(docEntry.uri)

            if (blk.isDownload && hasFile) {
                if (docEntry != null) resolvedCount++ else missingCount++
            }

            val isFlashable = blk.isDownload && hasFile
            steps.add(
                ParsedFlashStep(
                    id = "scatter_${idx + 1}_$cleanPart",
                    order = idx + 1,
                    command = if (hasFile) "fastboot flash $cleanPart ${blk.fileName}" else "fastboot getvar partition-size:$cleanPart",
                    targetPartition = cleanPart,
                    sourceName = if (hasFile) blk.fileName else "NONE (${blk.type.ifEmpty { "SYS" }})",
                    sourceUriString = docEntry?.uri?.toString(),
                    dataOffset = 0L,
                    dataSize = actualSize,
                    startAddressHex = blk.startAddr,
                    isSparse = isSparse,
                    isFlashable = isFlashable,
                    status = if (isFlashable) FlashStepStatus.READY else FlashStepStatus.SKIPPED,
                    enabled = isFlashable
                )
            )
        }

        val largeParts = steps.filter { it.isFlashable && it.dataSize > LARGE_PARTITION_THRESHOLD_BYTES }
            .map { it.targetPartition }

        val extras = buildList {
            if (configVersion.isNotEmpty()) add("Scatter Version: $configVersion")
            if (bootChannel.isNotEmpty()) add("Boot Channel: $bootChannel")
            add("Total Partitions: ${steps.size} (${steps.count { it.isFlashable }} downloadable)")
        }

        val fwInfo = FirmwareInfoSummary(
            formatLabel = "MediaTek Android Scatter",
            chipsetOrPlatform = platform,
            projectOrBoard = project,
            storageType = storage,
            sectorSizeBytes = if (storage.contains("UFS", ignoreCase = true)) 4096 else 512,
            isFolderPackage = folderFiles.isNotEmpty(),
            resolvedImageCount = resolvedCount,
            missingImageCount = missingCount,
            largePartitionNames = largeParts,
            extraMetadataLines = extras
        )
        return steps to fwInfo
    }

    /**
     * Parses Qualcomm Firehose rawprogram*.xml files (`<program ... />` tags).
     */
    fun parseRawprogramXmlContent(
        xmlList: List<Pair<String, String>>,
        folderFiles: Map<String, FolderDocEntry>,
        sparseChecker: (Uri) -> Boolean
    ): Pair<List<ParsedFlashStep>, FirmwareInfoSummary> {
        val steps = mutableListOf<ParsedFlashStep>()
        var detectedSectorSize = 4096
        val luns = mutableSetOf<String>()
        var resolvedCount = 0
        var missingCount = 0
        var order = 1

        xmlList.forEach { (xmlName, xmlContent) ->
            try {
                val factory = XmlPullParserFactory.newInstance()
                val parser = factory.newPullParser()
                parser.setInput(StringReader(xmlContent))
                var event = parser.eventType
                while (event != XmlPullParser.END_DOCUMENT) {
                    if (event == XmlPullParser.START_TAG &&
                        parser.name.equals("program", ignoreCase = true)
                    ) {
                        val label = parser.getAttributeValue(null, "label")?.trim().orEmpty()
                        val filename = parser.getAttributeValue(null, "filename")?.trim().orEmpty()
                        val sectorSzStr = parser.getAttributeValue(null, "SECTOR_SIZE_IN_BYTES")?.trim()
                        val numSectorsStr = parser.getAttributeValue(null, "num_partition_sectors")?.trim()
                        val startSectorStr = parser.getAttributeValue(null, "start_sector")?.trim().orEmpty()
                        val lun = parser.getAttributeValue(null, "physical_partition_number")?.trim().orEmpty()
                        val sparseAttr = parser.getAttributeValue(null, "sparse")?.trim().orEmpty()

                        val sectorSz = sectorSzStr?.toIntOrNull()?.takeIf { it > 0 } ?: 4096
                        detectedSectorSize = sectorSz
                        if (lun.isNotEmpty()) luns.add(lun)

                        val numSectors = numSectorsStr?.toLongOrNull()?.coerceAtLeast(0L) ?: 0L
                        val declaredBytes = numSectors * sectorSz.toLong()
                        val partName = label.ifEmpty {
                            inferPartition(filename.lowercase(Locale.US), "raw_$order")
                        }
                        val hasFile = filename.isNotBlank()
                        val docEntry = if (hasFile) resolveFolderFile(filename, folderFiles) else null
                        val actualBytes = docEntry?.sizeBytes?.takeIf { it > 0L } ?: declaredBytes
                        val isSparse = sparseAttr.equals("true", ignoreCase = true) ||
                            (docEntry != null && sparseChecker(docEntry.uri))

                        if (hasFile) {
                            if (docEntry != null) resolvedCount++ else missingCount++
                        }

                        val addrLabel = buildString {
                            if (lun.isNotEmpty()) append("LUN$lun ")
                            if (startSectorStr.isNotEmpty()) append("Sec:$startSectorStr")
                        }.trim()

                        steps.add(
                            ParsedFlashStep(
                                id = "rawxml_${order}_${partName}_${lun.ifEmpty { "0" }}",
                                order = order,
                                command = if (hasFile) "fastboot flash $partName $filename" else "fastboot getvar partition-size:$partName",
                                targetPartition = partName,
                                sourceName = if (hasFile) filename else "EMPTY ($xmlName)",
                                sourceUriString = docEntry?.uri?.toString(),
                                dataOffset = 0L,
                                dataSize = actualBytes,
                                startAddressHex = addrLabel,
                                isSparse = isSparse,
                                isFlashable = hasFile,
                                status = if (hasFile) FlashStepStatus.READY else FlashStepStatus.SKIPPED,
                                enabled = hasFile
                            )
                        )
                        order++
                    }
                    event = parser.next()
                }
            } catch (_: Exception) {
            }
        }

        val largeParts = steps.filter { it.isFlashable && it.dataSize > LARGE_PARTITION_THRESHOLD_BYTES }
            .map { it.targetPartition }
            .distinct()

        val storageGuess = if (detectedSectorSize >= 4096) "UFS (4096B Sector)" else "eMMC (512B Sector)"
        val fwInfo = FirmwareInfoSummary(
            formatLabel = "Qualcomm Firehose Rawprogram XML",
            chipsetOrPlatform = "Qualcomm Snapdragon",
            projectOrBoard = if (luns.isNotEmpty()) "LUNs: ${luns.sorted().joinToString(",")}" else "LUN 0",
            storageType = storageGuess,
            sectorSizeBytes = detectedSectorSize,
            isFolderPackage = folderFiles.isNotEmpty(),
            resolvedImageCount = resolvedCount,
            missingImageCount = missingCount,
            largePartitionNames = largeParts,
            extraMetadataLines = listOf(

    }

    /**
     * Parses Fastboot .bat / .cmd / .sh flash scripts (Xiaomi, Motorola, Pixel, Nothing).
     */
    fun parseScriptContent(
        scriptText: String,
        fileName: String,
        folderFiles: Map<String, FolderDocEntry>,
        sparseChecker: (Uri) -> Boolean
    ): Pair<List<ParsedFlashStep>, FirmwareInfoSummary> {
        val result = mutableListOf<ParsedFlashStep>()
        var detectedBoard = "Unknown"
        var resolvedCount = 0
        var missingCount = 0
        var order = 1

        val productRegex = Regex("""\^product:\s*\*?([A-Za-z0-9_-]+)""")
        val boardVarRegex = Regex("""(?:BOARD|PRODUCT|DEVICE)\s*=\s*["']?([A-Za-z0-9_-]+)["']?""", RegexOption.IGNORE_CASE)

        scriptText.lineSequence().forEach { rawLine ->
            val rawTrim = rawLine.trim()
            productRegex.find(rawTrim)?.groupValues?.getOrNull(1)?.let { detectedBoard = it }
            boardVarRegex.find(rawTrim)?.groupValues?.getOrNull(1)?.let {
                if (detectedBoard == "Unknown") detectedBoard = it
            }

            val line = rawTrim
                .replace("%~dp0", "")
                .replace("%*", "")
                .replace("\$ANDROID_PRODUCT_OUT/", "")
                .replace("\"${'$'}FASTBOOT\"", "fastboot")
                .replace("\\", "/")

            if (line.isNotEmpty() &&
                !line.startsWith("::") &&
                !line.startsWith("#") &&
                !line.startsWith("REM ", ignoreCase = true) &&
                !line.startsWith("@echo", ignoreCase = true) &&
                !line.startsWith("echo ", ignoreCase = true) &&
                !line.contains("getvar product", ignoreCase = true) &&
                (line.contains("fastboot", ignoreCase = true) || line.contains("adb", ignoreCase = true))
            ) {
                val cleanCmd = line
                    .replace(Regex("^.*fastboot(\\.exe)?", RegexOption.IGNORE_CASE), "fastboot")
                    .replace(Regex("\\s+"), " ")
                    .substringBefore("||")
                    .substringBefore("&&")
                    .trim()

                // Auto Reboot toggle in header manages final reboot cleanly
                if (cleanCmd.equals("fastboot reboot", ignoreCase = true)) {
                    return@forEach
                }

                val parts = cleanCmd.split(" ").map { it.trim('"', '\'') }
                val flashIdx = parts.indexOfFirst { it.equals("flash", ignoreCase = true) }
                val eraseIdx = parts.indexOfFirst { it.equals("erase", ignoreCase = true) }

                val targetPart = when {
                    flashIdx >= 0 && flashIdx + 1 < parts.size -> parts[flashIdx + 1]
                    eraseIdx >= 0 && eraseIdx + 1 < parts.size -> parts[eraseIdx + 1]
                    else -> "device"
                }

                val srcFile = if (flashIdx >= 0 && flashIdx + 2 < parts.size) {
                    parts[flashIdx + 2]
                } else {
                    cleanCmd
                }

                val isFlashCmd = flashIdx >= 0 && flashIdx + 2 < parts.size
                val docEntry = if (isFlashCmd) resolveFolderFile(srcFile, folderFiles) else null
                if (isFlashCmd) {
                    if (docEntry != null) resolvedCount++ else missingCount++
                }
                val isSparse = docEntry != null && sparseChecker(docEntry.uri)

                result.add(
                    ParsedFlashStep(
                        id = "script_$order",
                        order = order,
                        command = cleanCmd,
                        targetPartition = targetPart,
                        sourceName = srcFile,
                        sourceUriString = docEntry?.uri?.toString(),
                        dataOffset = 0L,
                        dataSize = docEntry?.sizeBytes ?: 0L,
                        startAddressHex = if (isFlashCmd) "SCRIPT #$order" else "CMD",
                        isSparse = isSparse,
                        isFlashable = true,
                        status = FlashStepStatus.READY,
                        enabled = true
                    )
                )
                order++
            }
        }

        val largeParts = result.filter { it.dataSize > LARGE_PARTITION_THRESHOLD_BYTES }
            .map { it.targetPartition }
            .distinct()

        val fwInfo = FirmwareInfoSummary(
            formatLabel = "Fastboot Flash Script",
            chipsetOrPlatform = "Fastboot Protocol",
            projectOrBoard = detectedBoard,
            storageType = if (largeParts.isNotEmpty()) "Dynamic / Large Partition Ready" else "Fastboot Script",
            isFolderPackage = folderFiles.isNotEmpty(),
            resolvedImageCount = resolvedCount,
            missingImageCount = missingCount,
            largePartitionNames = largeParts,
            extraMetadataLines = listOf(
                "Script File: $fileName",
                "Script Steps: ${result.size} operations"
            )
        )
        return result to fwInfo
    }

    /**
     * Uses FileChannel O(1) seeking on ParcelFileDescriptor to parse multi-gigabyte
     * Huawei UPDATE.APP packages and extract real Hardware ID / Build Date metadata.
     */
    private fun parseRealUpdateAppWithChannel(
        context: Context,
        uri: Uri
    ): Pair<List<ParsedFlashStep>, FirmwareInfoSummary> {
        val steps = mutableListOf<ParsedFlashStep>()
        var hwId = "Huawei HiSilicon / Qualcomm"
        var buildDate = ""
        var buildTime = ""

        try {
            context.contentResolver.openFileDescriptor(uri, "r")?.use { pfd ->
                FileInputStream(pfd.fileDescriptor).channel.use { channel ->
                    val fileSize = channel.size()
                    var pos = 0L
                    val buf = ByteBuffer.allocate(96).order(ByteOrder.LITTLE_ENDIAN)
                    val magicProbe = ByteBuffer.allocate(4)
                    var order = 1

                    while (pos + 96 <= fileSize && steps.size < 64) {
                        buf.clear()
                        val read = channel.read(buf, pos)
                        if (read < 96) break
                        buf.flip()

                        var magicIdx = -1
                        for (i in 0..92) {
                            if (buf.get(i) == 0x55.toByte() &&
                                buf.get(i + 1) == 0xAA.toByte() &&
                                buf.get(i + 2) == 0x5A.toByte() &&
                                buf.get(i + 3) == 0xA5.toByte()
                            ) {
                                magicIdx = i
                                break
                            }
                        }

                        if (magicIdx < 0) {
                            pos += 4L
                            continue
                        }
                        if (magicIdx > 0) {
                            pos += magicIdx.toLong()
                            continue
                        }

                        val headerSize = buf.getInt(4).toLong() and 0xFFFFFFFFL
                        val dataSize = buf.getInt(24).toLong() and 0xFFFFFFFFL

                        if (buildDate.isEmpty()) {
                            val hwBytes = ByteArray(8).also {
                                buf.position(12)
                                buf.get(it)
                            }
                            val dateBytes = ByteArray(16).also {
                                buf.position(28)
                                buf.get(it)
                            }
                            val timeBytes = ByteArray(16).also {
                                buf.position(44)
                                buf.get(it)
                            }
                            val h = String(hwBytes, Charsets.US_ASCII).substringBefore('\u0000').trim()
                            val d = String(dateBytes, Charsets.US_ASCII).substringBefore('\u0000').trim()
                            val t = String(timeBytes, Charsets.US_ASCII).substringBefore('\u0000').trim()
                            if (h.isNotEmpty()) hwId = h
                            if (d.isNotEmpty()) buildDate = d
                            if (t.isNotEmpty()) buildTime = t
                        }

                        val nameBytes = ByteArray(32)
                        buf.position(60)
                        buf.get(nameBytes)
                        val rawName = String(nameBytes, Charsets.US_ASCII)
                            .substringBefore('\u0000')
                            .trim()
                            .lowercase(Locale.US)

                        if (headerSize in 92..65536 &&
                            rawName.isNotEmpty() &&
                            rawName.all { it.isLetterOrDigit() || it == '_' || it == '-' }
                        ) {
                            val dataOffset = pos + headerSize
                            if (dataOffset + dataSize <= fileSize) {
                                magicProbe.clear()
                                val isSp = if (dataSize >= 4L && channel.read(magicProbe, dataOffset) == 4) {
                                    isSparseMagic(magicProbe.array())
                                } else {
                                    false
                                }

                                steps.add(
                                    ParsedFlashStep(
                                        id = "uapp_$order",
                                        order = order,
                                        command = "fastboot flash $rawName",
                                        targetPartition = rawName,
                                        sourceName = "${rawName.uppercase(Locale.US)}.IMG (@0x${dataOffset.toString(16)})",
                                        sourceUriString = uri.toString(),
                                        dataOffset = dataOffset,
                                        dataSize = dataSize,
                                        startAddressHex = "0x${dataOffset.toString(16)}",
                                        isSparse = isSp,
                                        isFlashable = true,
                                        status = FlashStepStatus.READY
                                    )
                                )
                                order++
                                val packetTotal = headerSize + dataSize
                                val rem = packetTotal % 4L
                                val alignedAdvance = if (rem == 0L) packetTotal else packetTotal + (4L - rem)
                                pos += alignedAdvance
                                continue
                            }
                        }
                        pos += 4L
                    }
                }
            }
        } catch (_: Exception) {
        }

        val largeParts = steps.filter { it.dataSize > LARGE_PARTITION_THRESHOLD_BYTES }
            .map { it.targetPartition }
        val extras = buildList {
            if (buildDate.isNotEmpty()) add("Build Date: $buildDate $buildTime".trim())
            add("Embedded Images: ${steps.size} partitions")
        }

        val fwInfo = FirmwareInfoSummary(
            formatLabel = "Huawei UPDATE.APP",
            chipsetOrPlatform = "Hardware ID: $hwId",
            projectOrBoard = "Huawei / Honor",
            storageType = "Container Package",
            isFolderPackage = false,
            resolvedImageCount = steps.size,
            missingImageCount = 0,
            largePartitionNames = largeParts,
            extraMetadataLines = extras
        )
        return steps to fwInfo
    }

    private fun parseRealZipEntries(context: Context, uri: Uri): List<ParsedFlashStep> {
        val steps = mutableListOf<ParsedFlashStep>()

                ZipInputStream(stream).use { zis ->

                            val lower = name.lowercase(Locale.US)
                            if (lower.endsWith(".img") || lower.endsWith(".bin") || lower == "payload.bin") {
                                val part = inferPartition(lower, "super")
                                val cmd = if (lower == "payload.bin") {
                                    "adb sideload $name"
                                } else {
                                    "fastboot flash $part $name"
                                }

                                    ParsedFlashStep(
                                        id = "zip_$order",
                                        order = order,

                                        sourceUriString = null,

                            }

        return try {

                var readTotal = 0
                while (readTotal < maxBytes) {
                    val r = stream.read(buf, readTotal, maxBytes - readTotal)
                    if (r <= 0) break
                    readTotal += r
                }
                buf.copyOf(readTotal)

                    if (r <= 0) break

    private fun isSparseMagic(header: ByteArray): Boolean {
        return header.size >= 4 &&
            header[0] == 0x3A.toByte() &&
            header[1] == 0xFF.toByte() &&
            header[2] == 0x26.toByte() &&
            header[3] == 0xED.toByte()
    }

    private fun parseSparseHeaderDescription(header: ByteArray): String? {
        if (header.size < 28 || !isSparseMagic(header)) return null
        val bb = ByteBuffer.wrap(header).order(ByteOrder.LITTLE_ENDIAN)
        bb.position(4)
        val major = bb.short.toInt() and 0xFFFF
        val minor = bb.short.toInt() and 0xFFFF
        bb.short // file_hdr_sz
        bb.short // chunk_hdr_sz
        val blkSz = bb.int.toLong() and 0xFFFFFFFFL
        val totalBlks = bb.int.toLong() and 0xFFFFFFFFL
        val totalChunks = bb.int.toLong() and 0xFFFFFFFFL
        return "Sparse v$major.$minor (${blkSz}B blk, $totalBlks blks, $totalChunks chunks)"
    }

    private fun checkUpdateAppMagic(context: Context, uri: Uri): Boolean {
        return try {
            context.contentResolver.openInputStream(uri)?.use { stream ->
                val probe = ByteArray(128)
                val read = stream.read(probe)
                if (read < 4) return false
                for (i in 0..(read - 4)) {
                    if (probe[i] == 0x55.toByte() &&
                        probe[i + 1] == 0xAA.toByte() &&
                        probe[i + 2] == 0x5A.toByte() &&
                        probe[i + 3] == 0xA5.toByte()
                    ) {
                        return true
                    }
                }
                false
            } ?: false
        } catch (_: Exception) {
            false
        }
    }

    fun inferPartition(lowerName: String, fallback: String = "boot"): String {
        return when {
            lowerName.contains("init_boot") -> "init_boot"
            lowerName.contains("vendor_boot") -> "vendor_boot"
            lowerName.contains("boot") -> "boot"
            lowerName.contains("vbmeta_system") -> "vbmeta_system"
            lowerName.contains("vbmeta_vendor") -> "vbmeta_vendor"
            lowerName.contains("vbmeta") -> "vbmeta"
            lowerName.contains("recovery") -> "recovery"
            lowerName.contains("super") -> "super"
            lowerName.contains("system_ext") -> "system_ext"
            lowerName.contains("system") -> "system"
            lowerName.contains("vendor") -> "vendor"
            lowerName.contains("product") -> "product"
            lowerName.contains("dtbo") -> "dtbo"
            lowerName.contains("modem") || lowerName.contains("non-hlos") -> "modem"
            lowerName.contains("persist") -> "persist"
            lowerName.contains("userdata") -> "userdata"
            lowerName.contains("metadata") -> "metadata"
            else -> fallback
        }
    }

    fun formatSize(bytes: Long): String {
        if (bytes < 0L) return "Unknown"
        if (bytes == 0L) return "0 B"
        val kb = 1024.0
        val mb = kb * 1024.0
        val gb = mb * 1024.0
        return when {
            bytes >= gb -> String.format(Locale.US, "%.2f GB", bytes / gb)
            bytes >= mb -> String.format(Locale.US, "%.2f MB", bytes / mb)
            bytes >= kb -> String.format(Locale.US, "%.1f KB", bytes / kb)
            else -> "$bytes B"
        }
    }

package com.example.data

import java.nio.ByteBuffer
import java.nio.ByteOrder
import java.nio.channels.FileChannel

/**
 * Zero-OOM Fastboot Sparse & Chunked Streaming Planner.
 *
 * Prevents OutOfMemoryError and bootloader buffer overflow crashes on multi-gigabyte
 * partitions (system, super, vendor, userdata) by:
 * 1. Splitting Android Sparse images (0xED26FF3A) strictly along chunk/block boundaries
 *    into valid sub-sparse packets (each <= bootloader max-download-size) with leading
 *    DONT_CARE (0xCAC3) seek chunks.
 * 2. Wrapping oversized Raw images (> max-download-size) into sparse sub-packets on-the-fly
 *    so the entire partition writes cleanly using only a 16 KB reusable streaming buffer.
 */
object SparseChunkPlanner {

    const val SPARSE_HEADER_MAGIC = 0xED26FF3A.toInt()
    const val SPARSE_HEADER_SIZE = 28
    const val CHUNK_HEADER_SIZE = 12

    const val CHUNK_TYPE_RAW: Short = 0xCAC1.toShort()
    const val CHUNK_TYPE_FILL: Short = 0xCAC2.toShort()
    const val CHUNK_TYPE_DONT_CARE: Short = 0xCAC3.toShort()
    const val CHUNK_TYPE_CRC32: Short = 0xCAC4.toShort()

    const val DEFAULT_SAFE_CHUNK_BYTES = 128L * 1024L * 1024L // 128 MB

    sealed class ChunkSlice {
        abstract val outputBlocks: Long
        abstract val packetBytesContribution: Long

        data class RawSlice(
            val fileOffset: Long,
            val readByteLength: Long,
            override val outputBlocks: Long,
            val blockAlignedByteLength: Long = readByteLength
        ) : ChunkSlice() {
            override val packetBytesContribution: Long
                get() = CHUNK_HEADER_SIZE + blockAlignedByteLength
        }

        data class FillSlice(
            val fillBytes4: ByteArray,
            override val outputBlocks: Long
        ) : ChunkSlice() {
            override val packetBytesContribution: Long
                get() = CHUNK_HEADER_SIZE + 4L
        }

        data class DontCareSlice(
            override val outputBlocks: Long
        ) : ChunkSlice() {
            override val packetBytesContribution: Long
                get() = CHUNK_HEADER_SIZE.toLong()
        }
    }

    data class SparseTransferPacket(
        val packetIndex: Int,
        val blockSize: Int,
        val leadingDontCareBlocks: Long,
        val slices: List<ChunkSlice>,
        val totalTransferBytes: Long,
        val totalOutputBlocks: Long
    ) {
        fun buildPacketPrefixHeader(): ByteArray {
            val totalChunks = (if (leadingDontCareBlocks > 0L) 1 else 0) + slices.size
            val prefixSize = SPARSE_HEADER_SIZE + if (leadingDontCareBlocks > 0L) CHUNK_HEADER_SIZE else 0
            val bb = ByteBuffer.allocate(prefixSize).order(ByteOrder.LITTLE_ENDIAN)

            // sparse_header_t (28 bytes)
            bb.putInt(SPARSE_HEADER_MAGIC)
            bb.putShort(1) // major_version
            bb.putShort(0) // minor_version
            bb.putShort(SPARSE_HEADER_SIZE.toShort())
            bb.putShort(CHUNK_HEADER_SIZE.toShort())
            bb.putInt(blockSize)
            bb.putInt(totalOutputBlocks.toInt())
            bb.putInt(totalChunks)
            bb.putInt(0) // image_checksum

            if (leadingDontCareBlocks > 0L) {
                // chunk_header_t: DONT_CARE seek offset
                bb.putShort(CHUNK_TYPE_DONT_CARE)
                bb.putShort(0)
                bb.putInt(leadingDontCareBlocks.toInt())
                bb.putInt(CHUNK_HEADER_SIZE)
            }
            return bb.array()
        }

        companion object {
            fun buildChunkHeader(type: Short, chunkBlocks: Long, totalBytes: Long): ByteArray {
                return ByteBuffer.allocate(CHUNK_HEADER_SIZE).order(ByteOrder.LITTLE_ENDIAN).apply {
                    putShort(type)
                    putShort(0)
                    putInt(chunkBlocks.toInt())
                    putInt(totalBytes.toInt())
                }.array()
            }
        }
    }

    /**
     * Plans zero-RAM chunk splits for an Android Sparse image stored inside [channel]
     * starting at [baseOffset] with length [streamSize].
     */
    fun planSparseSplits(
        channel: FileChannel,
        baseOffset: Long,
        streamSize: Long,
        maxPacketBytes: Long
    ): List<SparseTransferPacket> {
        val headerBuf = ByteBuffer.allocate(SPARSE_HEADER_SIZE).order(ByteOrder.LITTLE_ENDIAN)
        if (channel.read(headerBuf, baseOffset) < SPARSE_HEADER_SIZE) return emptyList()
        headerBuf.flip()

        val magic = headerBuf.int
        if (magic != SPARSE_HEADER_MAGIC) return emptyList()

        headerBuf.short // major
        headerBuf.short // minor
        val fileHdrSz = (headerBuf.short.toInt() and 0xFFFF).coerceAtLeast(SPARSE_HEADER_SIZE)
        val chunkHdrSz = (headerBuf.short.toInt() and 0xFFFF).coerceAtLeast(CHUNK_HEADER_SIZE)
        val blkSz = headerBuf.int
        val totalBlks = headerBuf.int.toLong() and 0xFFFFFFFFL
        val totalChunks = headerBuf.int.toLong() and 0xFFFFFFFFL
        if (blkSz <= 0 || totalChunks <= 0L) return emptyList()

        val minOverhead = SPARSE_HEADER_SIZE + CHUNK_HEADER_SIZE + CHUNK_HEADER_SIZE
        val safeLimit = maxPacketBytes.coerceAtLeast(minOverhead + blkSz.toLong())

        val packets = mutableListOf<SparseTransferPacket>()
        var currentSlices = mutableListOf<ChunkSlice>()
        var currentLeadingDontCare = 0L
        var currentPacketBytes = SPARSE_HEADER_SIZE.toLong()
        var currentPacketContentBlocks = 0L
        var globalOutputBlock = 0L

        fun flushPacket() {
            if (currentSlices.isEmpty()) return
            val hasUsefulData = currentSlices.any { it !is ChunkSlice.DontCareSlice }
            if (hasUsefulData) {
                packets.add(
                    SparseTransferPacket(
                        packetIndex = packets.size + 1,
                        blockSize = blkSz,
                        leadingDontCareBlocks = currentLeadingDontCare,
                        slices = currentSlices,
                        totalTransferBytes = currentPacketBytes,
                        totalOutputBlocks = currentLeadingDontCare + currentPacketContentBlocks
                    )
                )
            }
            currentSlices = mutableListOf()
            currentLeadingDontCare = globalOutputBlock
            currentPacketBytes = SPARSE_HEADER_SIZE.toLong() +
                if (currentLeadingDontCare > 0L) CHUNK_HEADER_SIZE.toLong() else 0L
            currentPacketContentBlocks = 0L
        }

        val chunkHdrBuf = ByteBuffer.allocate(chunkHdrSz).order(ByteOrder.LITTLE_ENDIAN)
        var cursor = baseOffset + fileHdrSz
        val endLimit = baseOffset + streamSize

        for (c in 0L until totalChunks) {
            if (cursor + chunkHdrSz > endLimit) break
            chunkHdrBuf.clear()
            if (channel.read(chunkHdrBuf, cursor) < chunkHdrSz) break
            chunkHdrBuf.flip()

            val chunkType = chunkHdrBuf.short
            chunkHdrBuf.short // reserved
            val chunkSzBlocks = chunkHdrBuf.int.toLong() and 0xFFFFFFFFL
            val chunkTotalBytes = chunkHdrBuf.int.toLong() and 0xFFFFFFFFL
            val payloadOffset = cursor + chunkHdrSz
            val payloadBytes = (chunkTotalBytes - chunkHdrSz).coerceAtLeast(0L)

            when (chunkType) {
                CHUNK_TYPE_RAW -> {
                    var remBlocks = chunkSzBlocks
                    var sliceOffset = payloadOffset
                    while (remBlocks > 0L) {
                        val availableForRawPayload = safeLimit - currentPacketBytes - CHUNK_HEADER_SIZE
                        if (availableForRawPayload < blkSz) {
                            flushPacket()
                            continue
                        }
                        val maxBlocksFit = (availableForRawPayload / blkSz).coerceAtLeast(1L)
                        val takeBlocks = minOf(remBlocks, maxBlocksFit)
                        val takeBytes = takeBlocks * blkSz.toLong()

                        val slice = ChunkSlice.RawSlice(
                            fileOffset = sliceOffset,
                            readByteLength = takeBytes,
                            outputBlocks = takeBlocks,
                            blockAlignedByteLength = takeBytes
                        )
                        currentSlices.add(slice)
                        currentPacketBytes += slice.packetBytesContribution
                        currentPacketContentBlocks += takeBlocks
                        globalOutputBlock += takeBlocks
                        sliceOffset += takeBytes
                        remBlocks -= takeBlocks

                        if (remBlocks > 0L) {
                            flushPacket()
                        }
                    }
                }

                CHUNK_TYPE_FILL -> {
                    val needed = CHUNK_HEADER_SIZE + 4L
                    if (currentPacketBytes + needed > safeLimit) {
                        flushPacket()
                    }
                    val fillBuf = ByteBuffer.allocate(4)
                    if (payloadBytes >= 4L) {
                        channel.read(fillBuf, payloadOffset)
                    }
                    val slice = ChunkSlice.FillSlice(
                        fillBytes4 = fillBuf.array(),
                        outputBlocks = chunkSzBlocks
                    )
                    currentSlices.add(slice)
                    currentPacketBytes += slice.packetBytesContribution
                    currentPacketContentBlocks += chunkSzBlocks
                    globalOutputBlock += chunkSzBlocks
                }

                CHUNK_TYPE_DONT_CARE -> {
                    if (currentSlices.isEmpty()) {
                        globalOutputBlock += chunkSzBlocks
                        currentLeadingDontCare = globalOutputBlock
                        currentPacketBytes = SPARSE_HEADER_SIZE.toLong() +
                            if (currentLeadingDontCare > 0L) CHUNK_HEADER_SIZE.toLong() else 0L
                    } else {
                        if (currentPacketBytes + CHUNK_HEADER_SIZE > safeLimit) {
                            flushPacket()
                            globalOutputBlock += chunkSzBlocks
                            currentLeadingDontCare = globalOutputBlock
                            currentPacketBytes = SPARSE_HEADER_SIZE.toLong() +
                                if (currentLeadingDontCare > 0L) CHUNK_HEADER_SIZE.toLong() else 0L
                        } else {
                            val slice = ChunkSlice.DontCareSlice(outputBlocks = chunkSzBlocks)
                            currentSlices.add(slice)
                            currentPacketBytes += slice.packetBytesContribution
                            currentPacketContentBlocks += chunkSzBlocks
                            globalOutputBlock += chunkSzBlocks
                        }
                    }
                }

                CHUNK_TYPE_CRC32 -> {
                    // CRC chunks cover 0 output blocks and become invalid across split packets; skip.
                }

                else -> {
                    // Unknown chunk type — safely advance block cursor if specified
                    globalOutputBlock += chunkSzBlocks
                }
            }

            cursor += chunkTotalBytes.coerceAtLeast(chunkHdrSz.toLong())
        }

        flushPacket()
        return packets
    }

    /**
     * Wraps an oversized non-sparse Raw partition image (e.g., raw super.img or system.img)
     * into block-aligned Sparse sub-packets <= [maxPacketBytes] so Fastboot can flash
     * arbitrarily large raw partitions without truncating or exhausting device RAM.
     */
    fun planRawToSparseSplits(
        baseOffset: Long,
        streamSize: Long,
        maxPacketBytes: Long,
        blockSize: Int = 4096
    ): List<SparseTransferPacket> {
        if (streamSize <= 0L) return emptyList()
        val overhead = (SPARSE_HEADER_SIZE + CHUNK_HEADER_SIZE + CHUNK_HEADER_SIZE).toLong()
        val safePayloadLimit = (maxPacketBytes - overhead).coerceAtLeast(blockSize.toLong())
        val maxBlocksPerPacket = (safePayloadLimit / blockSize).coerceAtLeast(1L)
        val maxBytesPerPacket = maxBlocksPerPacket * blockSize

        val packets = mutableListOf<SparseTransferPacket>()
        var offset = 0L
        var outputBlock = 0L

        while (offset < streamSize) {
            val remaining = streamSize - offset
            val readLen = minOf(remaining, maxBytesPerPacket)
            val blocks = (readLen + blockSize - 1L) / blockSize
            val alignedLen = blocks * blockSize

            val leadingDontCare = outputBlock
            val prefixBytes = SPARSE_HEADER_SIZE.toLong() +
                if (leadingDontCare > 0L) CHUNK_HEADER_SIZE.toLong() else 0L

            val slice = ChunkSlice.RawSlice(
                fileOffset = baseOffset + offset,
                readByteLength = readLen,
                outputBlocks = blocks,
                blockAlignedByteLength = alignedLen
            )
            packets.add(
                SparseTransferPacket(
                    packetIndex = packets.size + 1,
                    blockSize = blockSize,
                    leadingDontCareBlocks = leadingDontCare,
                    slices = listOf(slice),
                    totalTransferBytes = prefixBytes + slice.packetBytesContribution,
                    totalOutputBlocks = leadingDontCare + blocks
                )
            )
            offset += readLen
            outputBlock += blocks
        }
        return packets
    }
}

        private const val ADB_AUTH_SIGNATURE = 2
        private const val ADB_AUTH_RSAPUBLICKEY = 3

        private const val ADB_RSA_KEY_BITS = 2048
        private const val ADB_RSA_MODULUS_WORDS = ADB_RSA_KEY_BITS / 32 // 64
        private const val ADB_RSA_STRUCT_BYTES = 4 + 4 + (ADB_RSA_MODULUS_WORDS * 4) * 2 + 4 // 524

        // ASN.1 SHA-1 DigestInfo prefix for 20-byte ADB auth token
        private val SHA1_DIGEST_INFO_PREFIX = byteArrayOf(
            0x30, 0x21, 0x30, 0x09, 0x06, 0x05, 0x2B, 0x0E,
            0x03, 0x02, 0x1A, 0x05, 0x00, 0x04, 0x14
        )

        // Fastboot Max Single Download Chunk (256 MB)
        private const val MAX_FASTBOOT_CHUNK = 256L * 1024L * 1024L

        /**
         * Encodes a 2048-bit RSAPublicKey into Android's exact 524-byte little-endian
         * RSAPublicKey binary struct (libcrypto_utils android_pubkey_encode) so adbd
         * properly verifies and persists it to /data/misc/adb/adb_keys when "Always allow" is checked.
         */
        internal fun encodeAdbPublicKeyStruct(pub: RSAPublicKey): ByteArray {
            val n = pub.modulus
            val r32 = BigInteger.ONE.shiftLeft(32)
            val mask32 = r32.subtract(BigInteger.ONE)
            val n0inv = r32.subtract(n.modInverse(r32)).and(mask32).toLong().toInt()
            val rr = BigInteger.ONE.shiftLeft(ADB_RSA_KEY_BITS * 2).mod(n)

            val buf = ByteBuffer.allocate(ADB_RSA_STRUCT_BYTES).order(ByteOrder.LITTLE_ENDIAN)
            buf.putInt(ADB_RSA_MODULUS_WORDS)
            buf.putInt(n0inv)
            for (i in 0 until ADB_RSA_MODULUS_WORDS) {
                buf.putInt(n.shiftRight(i * 32).and(mask32).toLong().toInt())
            }
            for (i in 0 until ADB_RSA_MODULUS_WORDS) {
                buf.putInt(rr.shiftRight(i * 32).and(mask32).toLong().toInt())
            }
            buf.putInt(pub.publicExponent.toInt())
            return buf.array()
        }

        internal fun encodeAdbPublicKeyPayload(
            pub: RSAPublicKey,
            comment: String = "fastdeck@android"
        ): ByteArray {
            val structBytes = encodeAdbPublicKeyStruct(pub)
            val base64Pub = Base64.encodeToString(structBytes, Base64.NO_WRAP)
            return "$base64Pub $comment\u0000".toByteArray(StandardCharsets.US_ASCII)
        }

        internal fun signAdbTokenPayload(privateKey: PrivateKey, token: ByteArray): ByteArray {
            val token20 = token.copyOf(20)
            val digestInfo = SHA1_DIGEST_INFO_PREFIX + token20

            try {
                val sig = Signature.getInstance("NONEwithRSA")
                sig.initSign(privateKey)
                sig.update(digestInfo)
                val out = sig.sign()
                if (out.size == 256) return out
            } catch (_: Exception) {
            }

            return try {
                val cipher = Cipher.getInstance("RSA/ECB/PKCS1Padding")
                cipher.init(Cipher.ENCRYPT_MODE, privateKey)
                cipher.doFinal(digestInfo)
            } catch (_: Exception) {
                ByteArray(256)
            }
        }
    }

    /**

    fun executeFastbootWireCommands(
        deviceName: String,
        wireCommands: List<String>,
        verboseWireLogs: Boolean = false,
        onLog: (String, LogLevel) -> Unit
    ): Boolean {
        val manager = usbManager ?: run {
            onLog("USB Host unavailable.", LogLevel.FAIL)
            return false
        }
        val device = manager.deviceList[deviceName] ?: run {
            onLog("Device not attached.", LogLevel.FAIL)
            return false
        }
        if (!manager.hasPermission(device)) {
            onLog("USB permission required. Please allow access.", LogLevel.WARN)
            requestPermission(deviceName)
            return false
        }

        val triple = findBulkEndpoints(device, preferProtocol = 0x03) ?: run {
            onLog("No USB Bulk endpoints found.", LogLevel.FAIL)
            return false
        }
        val (iface, epIn, epOut) = triple
        val connection = manager.openDevice(device) ?: run {
            onLog("Failed to open USB connection.", LogLevel.FAIL)
            return false
        }

        var allOk = true
        try {
            if (!connection.claimInterface(iface, true)) {
                onLog("Failed to claim Fastboot interface.", LogLevel.FAIL)
                return false
            }

            for (wireCommand in wireCommands) {
                val payload = wireCommand.toByteArray(StandardCharsets.US_ASCII)
                val tx = connection.bulkTransfer(epOut, payload, payload.size, 4000)
                if (tx < 0) {
                    onLog("USB transfer failed ($tx).", LogLevel.FAIL)
                    return false
                }
                if (verboseWireLogs) {
                    onLog("TX ($tx B) -> $wireCommand", LogLevel.USB)
                }

                val timeoutMs = if (wireCommand.startsWith("erase:") || wireCommand.startsWith("flash:")) {
                    20000
                } else {
                    4000
                }
                val ok = readFastbootResponses(connection, epIn, timeoutMs, verboseWireLogs, onLog)
                if (!ok) {
                    allOk = false
                    break
                }
            }
        } catch (e: Exception) {
            onLog("USB Fastboot exception: ${e.message}", LogLevel.FAIL)
            allOk = false
        } finally {
            try {
                connection.releaseInterface(iface)
                connection.close()
            } catch (_: Exception) {
            }
        }
        return allOk
    }

    /**
     * Streams binary image data from a Uri (or UPDATE.APP byte offset) using ParcelFileDescriptor / FileChannel
     * over USB Bulk endpoints using Fastboot download + flash protocol.
     * Automatically splits multi-GB partitions (super, system, vendor) along Sparse chunk boundaries
     * using only a 16 KB reusable streaming buffer to prevent OutOfMemoryError and bootloader crashes.
     */
    fun flashBinaryStreamOverUsb(
        deviceName: String,
        uri: Uri,
        partition: String,
        dataOffset: Long,
        dataSize: Long,
        disableVerity: Boolean,
        verboseWireLogs: Boolean = false,
        isCancelled: () -> Boolean = { false },
        onChunkInfo: (chunkIndex: Int, totalChunks: Int, chunkBytes: Long) -> Unit = { _, _, _ -> },
        onProgress: (Float) -> Unit,
        onLog: (String, LogLevel) -> Unit
    ): Boolean {
        val manager = usbManager ?: run {
            onLog("USB Host unavailable.", LogLevel.FAIL)
            return false
        }
        val device = manager.deviceList[deviceName] ?: run {
            onLog("Fastboot device not attached.", LogLevel.FAIL)
            return false
        }
        if (!manager.hasPermission(device)) {
            onLog("USB permission required. Please allow access.", LogLevel.WARN)
            requestPermission(deviceName)
            return false
        }

        val triple = findBulkEndpoints(device, preferProtocol = 0x03) ?: run {
            onLog("No Bulk endpoints found.", LogLevel.FAIL)
            return false
        }
        val (iface, epIn, epOut) = triple
        val connection = manager.openDevice(device) ?: run {
            onLog("Unable to open USB connection.", LogLevel.FAIL)
            return false
        }

        try {
            if (!connection.claimInterface(iface, true)) {
                onLog("Unable to claim USB interface.", LogLevel.FAIL)
                return false
            }

            val pfd = context.contentResolver.openFileDescriptor(uri, "r") ?: run {
                onLog("Could not open selected file.", LogLevel.FAIL)
                return false
            }

            pfd.use { fd ->
                FileInputStream(fd.fileDescriptor).use { fis ->
                    val channel = fis.channel
                    val fileTotal = fd.statSize.takeIf { it > 0 } ?: channel.size()
                    val streamBytes = if (dataSize > 0L) {
                        dataSize.coerceAtMost(fileTotal - dataOffset)
                    } else {
                        (fileTotal - dataOffset).coerceAtLeast(0L)
                    }

                    if (streamBytes <= 0L) {
                        onLog("Selected file is empty.", LogLevel.FAIL)
                        return false
                    }

                    val bootloaderMax = queryFastbootMaxDownloadSize(connection, epIn, epOut)
                    val safeMaxChunk = minOf(
                        bootloaderMax ?: SparseChunkPlanner.DEFAULT_SAFE_CHUNK_BYTES,
                        SparseChunkPlanner.DEFAULT_SAFE_CHUNK_BYTES
                    ).coerceAtLeast(16L * 1024L * 1024L)

                    if (streamBytes <= safeMaxChunk) {
                        onChunkInfo(1, 1, streamBytes)
                        return streamSingleDownloadAndFlash(
                            connection = connection,
                            epIn = epIn,
                            epOut = epOut,
                            fis = fis,
                            channel = channel,
                            partition = partition,
                            dataOffset = dataOffset,
                            transferSize = streamBytes,
                            disableVerity = disableVerity,
                            verboseWireLogs = verboseWireLogs,
                            isCancelled = isCancelled,
                            onProgress = onProgress,
                            onLog = onLog
                        )
                    }

                    // Oversized partition (e.g. system.img / super.img): split safely into sparse sub-packets
                    val magicBuf = ByteBuffer.allocate(4).order(ByteOrder.LITTLE_ENDIAN)
                    val isSparse = channel.read(magicBuf, dataOffset) == 4 && run {
                        magicBuf.flip()
                        magicBuf.int == SparseChunkPlanner.SPARSE_HEADER_MAGIC
                    }

                    val packets = if (isSparse) {
                        SparseChunkPlanner.planSparseSplits(
                            channel = channel,
                            baseOffset = dataOffset,
                            streamSize = streamBytes,
                            maxPacketBytes = safeMaxChunk
                        )
                    } else {
                        SparseChunkPlanner.planRawToSparseSplits(
                            baseOffset = dataOffset,
                            streamSize = streamBytes,
                            maxPacketBytes = safeMaxChunk
                        )
                    }

                    if (packets.isEmpty()) {
                        onLog("Failed to build chunk plan for $partition.", LogLevel.FAIL)
                        return false
                    }

                    onLog(
                        "$partition (${FlashFileProcessor.formatSize(streamBytes)}) split into ${packets.size} safe chunks (${FlashFileProcessor.formatSize(safeMaxChunk)} max).",
                        LogLevel.INFO
                    )

                    val grandTotalBytes = packets.sumOf { it.totalTransferBytes }.coerceAtLeast(1L)
                    var grandSentBytes = 0L
                    val ioBuf = ByteArray(16384)

                    for (packet in packets) {
                        if (isCancelled()) {
                            onLog("Transfer aborted by user.", LogLevel.WARN)
                            return false
                        }
                        onChunkInfo(packet.packetIndex, packets.size, packet.totalTransferBytes)

                        if (!initiateFastbootDownload(
                                connection = connection,
                                epIn = epIn,
                                epOut = epOut,
                                transferSize = packet.totalTransferBytes,
                                verboseWireLogs = verboseWireLogs,
                                onLog = onLog
                            )
                        ) {
                            return false
                        }

                        val prefixHdr = packet.buildPacketPrefixHeader()
                        if (connection.bulkTransfer(epOut, prefixHdr, prefixHdr.size, 6000) < 0) {
                            onLog("USB header write failed on chunk ${packet.packetIndex}.", LogLevel.FAIL)
                            return false
                        }
                        grandSentBytes += prefixHdr.size

                        for (slice in packet.slices) {
                            if (isCancelled()) return false
                            when (slice) {
                                is SparseChunkPlanner.ChunkSlice.RawSlice -> {
                                    val chunkHdr = SparseChunkPlanner.SparseTransferPacket.buildChunkHeader(
                                        type = SparseChunkPlanner.CHUNK_TYPE_RAW,
                                        chunkBlocks = slice.outputBlocks,
                                        totalBytes = SparseChunkPlanner.CHUNK_HEADER_SIZE + slice.blockAlignedByteLength
                                    )
                                    if (connection.bulkTransfer(epOut, chunkHdr, chunkHdr.size, 6000) < 0) {
                                        onLog("USB chunk header write failed.", LogLevel.FAIL)
                                        return false
                                    }
                                    grandSentBytes += chunkHdr.size

                                    channel.position(slice.fileOffset)
                                    var sliceSent = 0L
                                    while (sliceSent < slice.readByteLength) {
                                        if (isCancelled()) return false
                                        val toRead = minOf(ioBuf.size.toLong(), slice.readByteLength - sliceSent).toInt()
                                        val r = fis.read(ioBuf, 0, toRead)
                                        if (r <= 0) {
                                            onLog("Unexpected EOF reading $partition.", LogLevel.FAIL)
                                            return false
                                        }
                                        val w = connection.bulkTransfer(epOut, ioBuf, r, 8000)
                                        if (w < 0) {
                                            onLog("USB write failed ($w) on chunk ${packet.packetIndex}.", LogLevel.FAIL)
                                            return false
                                        }
                                        sliceSent += w
                                        grandSentBytes += w
                                        onProgress((grandSentBytes.toFloat() / grandTotalBytes.toFloat()).coerceIn(0f, 1f))
                                    }

                                    var padRem = slice.blockAlignedByteLength - slice.readByteLength
                                    if (padRem > 0L) {
                                        val padBuf = ByteArray(padRem.toInt().coerceAtMost(4096))
                                        while (padRem > 0L) {
                                            val sendPad = minOf(padRem, padBuf.size.toLong()).toInt()
                                            if (connection.bulkTransfer(epOut, padBuf, sendPad, 4000) < 0) {
                                                return false
                                            }
                                            padRem -= sendPad
                                            grandSentBytes += sendPad
                                        }
                                    }
                                }

                                is SparseChunkPlanner.ChunkSlice.FillSlice -> {
                                    val chunkHdr = SparseChunkPlanner.SparseTransferPacket.buildChunkHeader(
                                        type = SparseChunkPlanner.CHUNK_TYPE_FILL,
                                        chunkBlocks = slice.outputBlocks,
                                        totalBytes = SparseChunkPlanner.CHUNK_HEADER_SIZE + 4L
                                    )
                                    val combined = chunkHdr + slice.fillBytes4
                                    if (connection.bulkTransfer(epOut, combined, combined.size, 6000) < 0) {
                                        return false
                                    }
                                    grandSentBytes += combined.size
                                }

                                is SparseChunkPlanner.ChunkSlice.DontCareSlice -> {
                                    val chunkHdr = SparseChunkPlanner.SparseTransferPacket.buildChunkHeader(
                                        type = SparseChunkPlanner.CHUNK_TYPE_DONT_CARE,
                                        chunkBlocks = slice.outputBlocks,
                                        totalBytes = SparseChunkPlanner.CHUNK_HEADER_SIZE.toLong()
                                    )
                                    if (connection.bulkTransfer(epOut, chunkHdr, chunkHdr.size, 6000) < 0) {
                                        return false
                                    }
                                    grandSentBytes += chunkHdr.size
                                }
                            }
                        }

                        if (!readFastbootResponses(connection, epIn, 15000, verboseWireLogs, onLog)) {
                            return false
                        }
                        if (!sendFastbootFlashCommand(connection, epIn, epOut, partition, 45000, verboseWireLogs, onLog)) {
                            return false
                        }
                        onLog(
                            "$partition chunk ${packet.packetIndex}/${packets.size} flashed OK.",
                            LogLevel.OKAY
                        )
                    }
                    onProgress(1f)
                    return true
                }
            }
        } catch (e: Exception) {
            onLog("Flash error: ${e.message}", LogLevel.FAIL)
            return false
        } finally {
            try {
                connection.releaseInterface(iface)
                connection.close()
            } catch (_: Exception) {
            }
        }
    }

    private fun queryFastbootMaxDownloadSize(
        connection: UsbDeviceConnection,
        epIn: UsbEndpoint,
        epOut: UsbEndpoint
    ): Long? {
        return try {
            val cmd = "getvar:max-download-size".toByteArray(StandardCharsets.US_ASCII)
            if (connection.bulkTransfer(epOut, cmd, cmd.size, 2500) < 0) return null
            val buf = ByteArray(128)
            val rx = connection.bulkTransfer(epIn, buf, buf.size, 2500)
            if (rx < 4) return null
            val resp = String(buf, 0, rx, StandardCharsets.US_ASCII).trim()
            if (!resp.startsWith("OKAY")) return null
            val valueStr = resp.removePrefix("OKAY").trim().lowercase(Locale.US)
            val parsed = if (valueStr.startsWith("0x")) {
                valueStr.removePrefix("0x").toLongOrNull(16)
            } else {
                valueStr.toLongOrNull()
            }
            parsed?.takeIf { it >= 16L * 1024L * 1024L }
        } catch (_: Exception) {
            null
        }
    }

    private fun initiateFastbootDownload(
        connection: UsbDeviceConnection,
        epIn: UsbEndpoint,
        epOut: UsbEndpoint,
        transferSize: Long,
        verboseWireLogs: Boolean,
        onLog: (String, LogLevel) -> Unit
    ): Boolean {
        val downloadCmd = String.format(Locale.US, "download:%08x", transferSize)
        val cmdBytes = downloadCmd.toByteArray(StandardCharsets.US_ASCII)
        val txCmd = connection.bulkTransfer(epOut, cmdBytes, cmdBytes.size, 4000)
        if (txCmd < 0) {
            onLog("Failed to initiate transfer ($txCmd).", LogLevel.FAIL)
            return false
        }
        if (verboseWireLogs) {
            onLog("TX -> $downloadCmd (${FlashFileProcessor.formatSize(transferSize)})", LogLevel.USB)
        }

        val respBuf = ByteArray(64)
        val rx = connection.bulkTransfer(epIn, respBuf, respBuf.size, 5000)
        if (rx <= 0) {
            onLog("No response from bootloader.", LogLevel.FAIL)
            return false
        }
        val reply = String(respBuf, 0, rx, StandardCharsets.US_ASCII).trim()
        if (verboseWireLogs) {
            onLog("RX <- $reply", LogLevel.USB)
        }
        if (!reply.startsWith("DATA")) {
            onLog("Bootloader rejected transfer: $reply", LogLevel.FAIL)
            return false
        }
        return true
    }

    private fun sendFastbootFlashCommand(
        connection: UsbDeviceConnection,
        epIn: UsbEndpoint,
        epOut: UsbEndpoint,
        partition: String,
        timeoutMs: Int,
        verboseWireLogs: Boolean,
        onLog: (String, LogLevel) -> Unit
    ): Boolean {
        val flashCmd = "flash:$partition"
        val flashBytes = flashCmd.toByteArray(StandardCharsets.US_ASCII)
        val txFlash = connection.bulkTransfer(epOut, flashBytes, flashBytes.size, 4000)
        if (txFlash < 0) {
            onLog("Flash transfer failed ($txFlash).", LogLevel.FAIL)
            return false

        if (verboseWireLogs) {
            onLog("TX -> $flashCmd", LogLevel.USB)

    private fun streamSingleDownloadAndFlash(
        connection: UsbDeviceConnection,
        epIn: UsbEndpoint,
        epOut: UsbEndpoint,
        fis: FileInputStream,
        channel: java.nio.channels.FileChannel,
        partition: String,
        dataOffset: Long,
        transferSize: Long,
        disableVerity: Boolean,
        verboseWireLogs: Boolean,
        isCancelled: () -> Boolean,
        onProgress: (Float) -> Unit,
        onLog: (String, LogLevel) -> Unit
    ): Boolean {
        if (!initiateFastbootDownload(connection, epIn, epOut, transferSize, verboseWireLogs, onLog)) {
            return false
        }

        channel.position(dataOffset)
        val chunk = ByteArray(16384)
        var sentTotal = 0L
        while (sentTotal < transferSize) {
            if (isCancelled()) {
                onLog("Transfer aborted by user.", LogLevel.WARN)
                return false
            }
            val toRead = minOf(chunk.size.toLong(), transferSize - sentTotal).toInt()
            val r = fis.read(chunk, 0, toRead)
            if (r <= 0) break

            // Patch vbmeta disable-verity flag at AVBHeader offset 123 if requested
            if (sentTotal == 0L && disableVerity && partition.startsWith("vbmeta") && r >= 128) {
                if (chunk[0] == 'A'.code.toByte() &&
                    chunk[1] == 'V'.code.toByte() &&
                    chunk[2] == 'B'.code.toByte() &&
                    chunk[3] == '0'.code.toByte()
                ) {
                    chunk[123] = (chunk[123].toInt() or 0x03).toByte()
                    onLog("AVB0 verity flags patched.", LogLevel.OKAY)
                }
            }

            val written = connection.bulkTransfer(epOut, chunk, r, 6000)
            if (written < 0) {
                onLog("USB write failed ($written).", LogLevel.FAIL)
                return false
            }
            sentTotal += written
            onProgress((sentTotal.toFloat() / transferSize.toFloat()).coerceIn(0f, 1f))
        }

        onLog("Sent ${FlashFileProcessor.formatSize(sentTotal)} to $partition.", LogLevel.OKAY)
        if (!readFastbootResponses(connection, epIn, 10000, verboseWireLogs, onLog)) {
            return false
        }
        return sendFastbootFlashCommand(connection, epIn, epOut, partition, 30000, verboseWireLogs, onLog)
    }

    /**
     * Native USB ADB Protocol Execution (A_CNXN -> A_AUTH RSA -> A_OPEN -> A_WRTE -> A_CLSE).
     * Executes real ADB reboot or shell commands over USB OTG when an ADB device is connected!
     */
    fun executeAdbOverUsbOtg(
        deviceName: String,
        command: String,
        verboseWireLogs: Boolean = false,
        onLog: (String, LogLevel) -> Unit
    ) {
        val manager = usbManager ?: return
        val device = manager.deviceList[deviceName] ?: run {
            onLog("No USB OTG device connected.", LogLevel.FAIL)
            return
        }
        if (!manager.hasPermission(device)) {
            onLog("USB permission required. Please allow access.", LogLevel.WARN)
            requestPermission(deviceName)
            return
        }
        val triple = findBulkEndpoints(device, preferProtocol = 0x01) ?: run {
            onLog("No Bulk endpoints found.", LogLevel.FAIL)
            return
        }
        val (iface, epIn, epOut) = triple
        val connection = manager.openDevice(device) ?: run {
            onLog("Failed to open USB connection.", LogLevel.FAIL)
            return
        }

        try {
            if (!connection.claimInterface(iface, true)) {
                onLog("Cannot claim ADB interface.", LogLevel.FAIL)
                return
            }

            // Step 1: Send A_CNXN
            val banner = "host::features=shell_v2,cmd,stat_v2,abb_exec,fastbootd\u0000"
                .toByteArray(StandardCharsets.UTF_8)
            sendAdbPacket(connection, epOut, A_CNXN, A_VERSION, MAX_ADB_DATA, banner)
            if (verboseWireLogs) {
                onLog("ADB USB TX -> CNXN (${device.productName})", LogLevel.USB)
            }

            // Step 2: Handle A_AUTH or A_CNXN response
            var connected = false
            var sentSignature = false
            var sentPublicKey = false
            var authWaitRetries = 0
            val keyPair = getOrCreateAdbRsaKeyPair()

            for (attempt in 0 until 12) {
                val readTimeout = if (sentPublicKey) 4000 else 3500
                val pkt = readAdbPacket(connection, epIn, readTimeout)
                if (pkt == null) {
                    if (sentPublicKey && authWaitRetries < 3) {
                        authWaitRetries++
                        continue
                    }
                    break
                }
                when (pkt.command) {
                    A_CNXN -> {
                        val remoteBanner = String(pkt.payload, StandardCharsets.UTF_8)
                            .trim { it <= ' ' || it == '\u0000' }
                        if (verboseWireLogs) {
                            onLog("ADB USB Connected: $remoteBanner", LogLevel.OKAY)
                        }
                        connected = true
                        break
                    }
                    A_AUTH -> {
                        if (pkt.arg0 == ADB_AUTH_TOKEN) {
                            if (!sentSignature) {
                                val sig = signAdbToken(keyPair, pkt.payload)
                                sendAdbPacket(connection, epOut, A_AUTH, ADB_AUTH_SIGNATURE, 0, sig)
                                sentSignature = true
                                if (verboseWireLogs) {
                                    onLog("ADB USB TX -> AUTH_SIGNATURE (2048-bit RSA)", LogLevel.USB)
                                }
                            } else if (!sentPublicKey) {
                                val pubKeyBytes = encodeAdbPublicKey(keyPair)
                                sendAdbPacket(connection, epOut, A_AUTH, ADB_AUTH_RSAPUBLICKEY, 0, pubKeyBytes)
                                sentPublicKey = true
                                onLog(
                                    "Tap 'Allow USB Debugging' on target phone.",
                                    LogLevel.WARN
                                )
                            }
                        }
                    }
                }
            }

            if (!connected) {
                onLog("Waiting for USB Debugging authorization on phone.", LogLevel.WARN)
                return
            }

            // Route ADB host/service commands cleanly without falling into remote /system/bin/sh
            val trimmedCmd = command.trim()
            val lower = trimmedCmd.lowercase(Locale.US)
            if (lower == "adb devices" || lower == "adb devices -l") {
                onLog("${device.serialNumber ?: device.deviceName}\tdevice product:${device.productName}", LogLevel.OKAY)
                return
            }
            if (lower == "adb get-state") {
                onLog("device", LogLevel.OKAY)
                return
            }
            if (lower == "adb get-serialno") {
                onLog(device.serialNumber ?: device.deviceName, LogLevel.OKAY)
                return
            }

            val serviceDest: String = when {
                lower == "adb" -> {
                    onLog("Usage: adb devices | adb shell <cmd> | adb reboot [bootloader|recovery]", LogLevel.INFO)
                    return
                }
                lower == "adb reboot" -> "reboot:\u0000"
                lower.startsWith("adb reboot ") -> {
                    val target = trimmedCmd.substringAfter("reboot ", "").trim()
                    "reboot:$target\u0000"
                }
                lower == "adb root" -> "root:\u0000"
                lower == "adb unroot" -> "unroot:\u0000"
                lower == "adb remount" -> "remount:\u0000"
                lower == "adb usb" -> "usb:\u0000"
                lower.startsWith("adb tcpip ") -> {
                    val port = trimmedCmd.substringAfter("tcpip ", "").trim()
                    "tcpip:$port\u0000"
                }
                lower == "adb shell" -> {
                    onLog("Usage: adb shell <command>", LogLevel.INFO)
                    return
                }
                lower.startsWith("adb shell ") -> {
                    val shellCmd = trimmedCmd.substring(10).trim()
                    "shell:$shellCmd\u0000"
                }
                lower.startsWith("adb ") -> {
                    val sub = trimmedCmd.substring(4).trim()
                    onLog(
                        "adb: unsupported subcommand '$sub'. Use 'adb shell <cmd>', 'adb reboot [target]', or 'adb devices'.",
                        LogLevel.FAIL
                    )
                    return
                }
                lower.startsWith("shell ") -> {
                    val shellCmd = trimmedCmd.substring(6).trim()
                    "shell:$shellCmd\u0000"
                }
                else -> "shell:$trimmedCmd\u0000"
            }

            val isRebootService = serviceDest.startsWith("reboot:")
            val localId = 1
            sendAdbPacket(
                connection,
                epOut,
                A_OPEN,
                localId,
                0,
                serviceDest.toByteArray(StandardCharsets.UTF_8)
            )
            if (verboseWireLogs) {
                onLog("ADB USB TX -> OPEN '${serviceDest.trimEnd('\u0000')}'", LogLevel.USB)
            }

            // Step 4: Read A_OKAY / A_WRTE / A_CLSE stream output
            var remoteId = 0
            var streamOpened = false
            for (loop in 0 until 128) {
                val pktTimeout = if (isRebootService && streamOpened) 1200 else 3500
                val pkt = readAdbPacket(connection, epIn, pktTimeout) ?: break
                when (pkt.command) {
                    A_OKAY -> {
                        remoteId = pkt.arg0
                        streamOpened = true
                    }
                    A_WRTE -> {
                        remoteId = pkt.arg0
                        val text = String(pkt.payload, StandardCharsets.UTF_8)
                        text.lines().filter { it.isNotBlank() }.forEach { line ->
                            onLog(line, LogLevel.INFO)
                        }
                        sendAdbPacket(connection, epOut, A_OKAY, localId, remoteId, ByteArray(0))
                    }
                    A_CLSE -> {
                        sendAdbPacket(connection, epOut, A_CLSE, localId, remoteId, ByteArray(0))
                        if (verboseWireLogs) {
                            onLog("ADB stream closed (OKAY).", LogLevel.OKAY)
                        }
                        break
                    }
                }
            }
            if (isRebootService && streamOpened) {
                onLog("Reboot command accepted.", LogLevel.OKAY)
            }
        } catch (e: Exception) {
            onLog("ADB USB error: ${e.message}", LogLevel.FAIL)
        } finally {
            try {
                connection.releaseInterface(iface)
                connection.close()
            } catch (_: Exception) {
            }
        }
    }

    /**
     * Executes a real local shell command via ProcessBuilder when inspecting local host properties.
     */
    fun executeLocalShell(command: String, onLog: (String, LogLevel) -> Unit) {
        try {
            val clean = command
                .removePrefix("adb shell ")
                .removePrefix("shell ")
                .trim()
            val process = ProcessBuilder("sh", "-c", clean)
                .redirectErrorStream(true)
                .start()

            var lineCount = 0
            BufferedReader(InputStreamReader(process.inputStream)).useLines { lines ->
                lines.take(100).forEach { line ->
                    if (line.isNotBlank()) {
                        onLog(line, LogLevel.INFO)
                        lineCount++
                    }
                }
            }
            val exitCode = process.waitFor()
            if (exitCode != 0) {
                onLog("Command failed (exit $exitCode) — connect target via USB OTG.", LogLevel.WARN)
            }
        } catch (e: Exception) {
            onLog("Execution error: ${e.message}", LogLevel.FAIL)
        }
    }

    private fun readFastbootResponses(
        connection: UsbDeviceConnection,
        epIn: UsbEndpoint,
        timeoutMs: Int,
        verboseWireLogs: Boolean,
        onLog: (String, LogLevel) -> Unit
    ): Boolean {
        val buf = ByteArray(256)
        var polls = 0
        while (polls < 64) {
            polls++
            val rx = connection.bulkTransfer(epIn, buf, buf.size, timeoutMs)
            if (rx < 4) {
                onLog("Fastboot response timeout.", LogLevel.WARN)
                return false
            }
            val status = String(buf, 0, 4, StandardCharsets.US_ASCII)
            val body = if (rx > 4) {
                String(buf, 4, rx - 4, StandardCharsets.US_ASCII).trim()
            } else {
                ""
            }
            when (status) {
                "INFO" -> onLog(body.ifEmpty { status }, LogLevel.INFO)
                "TEXT" -> onLog(body, LogLevel.INFO)
                "OKAY" -> {
                    onLog("OKAY ${if (body.isNotEmpty()) "[$body]" else ""}".trim(), LogLevel.OKAY)
                    return true
                }
                "FAIL" -> {
                    onLog("FAILED: $body", LogLevel.FAIL)
                    return false
                }
                else -> {
                    if (verboseWireLogs) {
                        onLog("RX: $status$body", LogLevel.USB)
                    }
                    return true
                }
            }
        }
        return false
    }

    private class AdbPacket(
        val command: Int,
        val arg0: Int,
        val arg1: Int,
        val payload: ByteArray
    )

    private fun sendAdbPacket(
        connection: UsbDeviceConnection,
        epOut: UsbEndpoint,
        command: Int,
        arg0: Int,
        arg1: Int,
        payload: ByteArray
    ) {
        var checksum = 0
        for (b in payload) {
            checksum += (b.toInt() and 0xFF)
        }
        val header = ByteBuffer.allocate(24).order(ByteOrder.LITTLE_ENDIAN).apply {
            putInt(command)
            putInt(arg0)
            putInt(arg1)
            putInt(payload.size)
            putInt(checksum)
            putInt(command xor -1)
        }.array()

        connection.bulkTransfer(epOut, header, header.size, 2500)
        if (payload.isNotEmpty()) {
            connection.bulkTransfer(epOut, payload, payload.size, 3000)
        }
    }

    private fun readAdbPacket(
        connection: UsbDeviceConnection,
        epIn: UsbEndpoint,
        timeoutMs: Int
    ): AdbPacket? {
        val hdr = ByteArray(24)
        var rx = -1
        for (zlpRetry in 0 until 3) {
            rx = connection.bulkTransfer(epIn, hdr, hdr.size, timeoutMs)
            if (rx != 0) break
        }
        if (rx != 24) return null

        val bb = ByteBuffer.wrap(hdr).order(ByteOrder.LITTLE_ENDIAN)
        val cmd = bb.int
        val arg0 = bb.int
        val arg1 = bb.int
        val len = bb.int
        bb.int // checksum
        bb.int // magic

        val payload = if (len in 1..MAX_ADB_DATA) {
            val data = ByteArray(len)
            var total = 0
            var zeroReads = 0
            while (total < len) {
                val r = connection.bulkTransfer(epIn, data, total, len - total, timeoutMs)
                if (r < 0) break
                if (r == 0) {
                    zeroReads++
                    if (zeroReads >= 3) break
                    continue
                }
                zeroReads = 0
                total += r
            }
            data
        } else {
            ByteArray(0)
        }
        return AdbPacket(cmd, arg0, arg1, payload)
    }

    private fun getOrCreateAdbRsaKeyPair(): KeyPair {
        val privFile = File(context.filesDir, "adb_rsa_priv.key")
        val pubFile = File(context.filesDir, "adb_rsa_pub.key")
        if (privFile.exists() && pubFile.exists()) {
            try {
                val kf = KeyFactory.getInstance("RSA")
                val priv = kf.generatePrivate(PKCS8EncodedKeySpec(privFile.readBytes()))
                val pub = kf.generatePublic(X509EncodedKeySpec(pubFile.readBytes()))
                if (pub is RSAPublicKey && priv is RSAPrivateKey && pub.modulus.bitLength() == ADB_RSA_KEY_BITS) {
                    return KeyPair(pub, priv)
                }
            } catch (_: Exception) {
            }
        }
        val kpg = KeyPairGenerator.getInstance("RSA")
        kpg.initialize(ADB_RSA_KEY_BITS)
        val kp = kpg.generateKeyPair()
        try {
            privFile.writeBytes(kp.private.encoded)
            pubFile.writeBytes(kp.public.encoded)
        } catch (_: Exception) {
        }
        return kp
    }

    private fun signAdbToken(keyPair: KeyPair, token: ByteArray): ByteArray {
        return signAdbTokenPayload(keyPair.private, token)
    }

    private fun encodeAdbPublicKey(keyPair: KeyPair): ByteArray {
        val rsaPub = keyPair.public as? RSAPublicKey
            ?: throw IllegalStateException("ADB RSA public key required")
        return encodeAdbPublicKeyPayload(rsaPub)
    }

    private fun inspectDevice(manager: UsbManager, device: UsbDevice): UsbPortInfo {
        val hasPerm = try {
            manager.hasPermission(device)
        } catch (_: Exception) {
            false
        }

        var protoType = UsbProtocolType.VENDOR_PORT
        var ifaceId = 0
        var epInAddr = 0
        var epOutAddr = 0
        var maxPkt = 512

        for (i in 0 until device.interfaceCount) {
            val iface = device.getInterface(i)
            if (iface.interfaceClass == 0xFF && iface.interfaceSubclass == 0x42) {
                if (iface.interfaceProtocol == 0x01) {
                    protoType = UsbProtocolType.ADB
                    ifaceId = iface.id
                } else if (iface.interfaceProtocol == 0x03) {
                    protoType = UsbProtocolType.FASTBOOT
                    ifaceId = iface.id
                }
            }
            for (e in 0 until iface.endpointCount) {
                val ep = iface.getEndpoint(e)
                if (ep.type == UsbConstants.USB_ENDPOINT_XFER_BULK) {
                    maxPkt = ep.maxPacketSize
                    if (ep.direction == UsbConstants.USB_DIR_IN) epInAddr = ep.address
                    if (ep.direction == UsbConstants.USB_DIR_OUT) epOutAddr = ep.address
                }
            }
        }

        val prodName = try {
            device.productName ?: "USB Device"
        } catch (_: Exception) {
            "USB Device"
        }

        val serial = try {
            if (hasPerm) device.serialNumber ?: "N/A" else "Perm Required"
        } catch (_: Exception) {
            "Perm Required"
        }

        return UsbPortInfo(
            deviceName = device.deviceName,
            vendorId = device.vendorId,
            productId = device.productId,
            vendorHex = String.format(Locale.US, "0x%04X", device.vendorId),
            productHex = String.format(Locale.US, "0x%04X", device.productId),
            productName = prodName,
            serialNumber = serial,
            interfaceId = ifaceId,
            endpointInAddress = epInAddr,
            endpointOutAddress = epOutAddr,
            maxPacketSize = maxPkt,
            hasPermission = hasPerm,
            protocolType = protoType
        )
    }

    private fun findBulkEndpoints(
        device: UsbDevice,
        preferProtocol: Int
    ): Triple<UsbInterface, UsbEndpoint, UsbEndpoint>? {
        var fallback: Triple<UsbInterface, UsbEndpoint, UsbEndpoint>? = null
        for (i in 0 until device.interfaceCount) {
            val iface = device.getInterface(i)
            var epIn: UsbEndpoint? = null
            var epOut: UsbEndpoint? = null
            for (e in 0 until iface.endpointCount) {
                val ep = iface.getEndpoint(e)
                if (ep.type == UsbConstants.USB_ENDPOINT_XFER_BULK) {
                    if (ep.direction == UsbConstants.USB_DIR_IN) epIn = ep
                    if (ep.direction == UsbConstants.USB_DIR_OUT) epOut = ep
                }
            }
            if (epIn != null && epOut != null) {
                if (iface.interfaceClass == 0xFF &&
                    iface.interfaceSubclass == 0x42 &&
                    iface.interfaceProtocol == preferProtocol
                ) {
                    return Triple(iface, epIn, epOut)
                }
                if (fallback == null) {
                    fallback = Triple(iface, epIn, epOut)
                }
            }
        }
        return fallback
    }
}

import java.text.SimpleDateFormat
import java.util.Date
import java.util.Locale
import java.util.concurrent.atomic.AtomicLong

data class ServiceUiState(
    val themeMode: ThemeMode = ThemeMode.DARK,
    val themeDesign: ThemeDesign = ThemeDesign.CYBER_NEON,
    val fontSize: AppFontSize = AppFontSize.MEDIUM,
    val soundEnabled: Boolean = true,
    val screenMode: ToolScreenMode = ToolScreenMode.FASTBOOT_INFO,
    val usbPorts: List<UsbPortInfo> = emptyList(),
    val selectedPortName: String? = null,
    val selectedBrand: ServiceBrand = ServiceBrand.UNIVERSAL,
    val selectedAction: ServiceActionItem = CommandRepository.allActions.first {
        !it.isAdb && it.group == ActionGroup.INFO
    },
    val selectedPartition: String = CommandRepository.supportedPartitions.first(),
    val disableVerity: Boolean = true,
    val clearUserdata: Boolean = false,
    val autoReboot: Boolean = true,
    val checksumVerify: Boolean = true,
    val useFastbootd: Boolean = false,
    val selectedFlashFile: SelectedFlashFile? = null,
    val isBusy: Boolean = false,
    val statusBarText: String = "USB OTG Ready",
    val progressPercent: Float = 0f,
    val currentFlashingPartition: String = "",
    val currentChunkLabel: String = "",
    val consoleLogs: List<ConsoleLogEntry> = emptyList(),
    val terminalInput: String = "",
    val terminalLines: List<TerminalLine> = emptyList(),
    val terminalHistory: List<String> = emptyList()
)

class ServiceViewModel(application: Application) : AndroidViewModel(application) {

    private val prefs = application.applicationContext.getSharedPreferences(
        "fastdeck_otg_prefs",
        Context.MODE_PRIVATE
    )
    private val usbController = UsbPortController(application.applicationContext)
    private val soundManager = SoundEffectManager()
    private val logSequence = AtomicLong(1L)
    private val terminalSequence = AtomicLong(1L)

    private var activeOperationJob: Job? = null
    private var hasCompletedInitialUsbScan = false
    private var knownPortsByName: Map<String, UsbPortInfo> = emptyMap()

    private val _uiState = MutableStateFlow(loadInitialState())
    val uiState: StateFlow<ServiceUiState> = _uiState.asStateFlow()

    init {
        soundManager.isEnabled = _uiState.value.soundEnabled
        scanUsbPorts(userInitiated = false)
    }

    private fun loadInitialState(): ServiceUiState {
        val savedMode = prefs.getString("theme_mode", ThemeMode.DARK.name)
            ?.let { runCatching { ThemeMode.valueOf(it) }.getOrNull() }
            ?: ThemeMode.DARK
        val savedDesign = prefs.getString("theme_design", ThemeDesign.CYBER_NEON.name)
            ?.let { runCatching { ThemeDesign.valueOf(it) }.getOrNull() }
            ?: ThemeDesign.CYBER_NEON
        val savedFont = prefs.getString("font_size", AppFontSize.MEDIUM.name)
            ?.let { runCatching { AppFontSize.valueOf(it) }.getOrNull() }
            ?: AppFontSize.MEDIUM
        val savedSound = prefs.getBoolean("sound_enabled", true)

        return ServiceUiState(
            themeMode = savedMode,
            themeDesign = savedDesign,
            fontSize = savedFont,
            soundEnabled = savedSound
        )
    }

    /**
     * Strictly filters actions by active Drawer sub-mode (ADB/Fastboot + Info/Unlock/Service)
     * and selected Brand so every section stays focused and uncluttered.
     */
    fun getAvailableActions(
        mode: ToolScreenMode = _uiState.value.screenMode,
        brand: ServiceBrand = _uiState.value.selectedBrand
    ): List<ServiceActionItem> {
        val modeFiltered = when (mode) {
            ToolScreenMode.ADB_INFO -> CommandRepository.allActions.filter {
                it.isAdb && it.group == ActionGroup.INFO
            }
            ToolScreenMode.ADB_UNLOCK -> CommandRepository.allActions.filter {
                it.isAdb && it.group == ActionGroup.UNLOCK
            }
            ToolScreenMode.ADB_SERVICE -> CommandRepository.allActions.filter {
                it.isAdb && it.group == ActionGroup.SERVICE
            }
            ToolScreenMode.FASTBOOT_INFO -> CommandRepository.allActions.filter {
                !it.isAdb && it.group == ActionGroup.INFO
            }
            ToolScreenMode.FASTBOOT_UNLOCK -> CommandRepository.allActions.filter {
                !it.isAdb && it.group == ActionGroup.UNLOCK
            }
            ToolScreenMode.FASTBOOT_SERVICE,
            ToolScreenMode.FASTBOOT_FLASH -> CommandRepository.allActions.filter {
                !it.isAdb && it.group == ActionGroup.SERVICE
            }
            ToolScreenMode.CMD_TERMINAL,
            ToolScreenMode.SETTINGS,
            ToolScreenMode.ABOUT -> emptyList()
        }
        val brandFiltered = modeFiltered.filter { it.brand == brand }
        return brandFiltered.ifEmpty {
            modeFiltered.filter { it.brand == ServiceBrand.UNIVERSAL }.ifEmpty { modeFiltered }
        }
    }

    fun toggleTheme() {
        val next = if (_uiState.value.themeMode == ThemeMode.DARK) ThemeMode.LIGHT else ThemeMode.DARK
        setThemeMode(next)
    }

    fun setThemeMode(mode: ThemeMode) {
        prefs.edit().putString("theme_mode", mode.name).apply()
        _uiState.update { it.copy(themeMode = mode) }
    }

    fun setThemeDesign(design: ThemeDesign) {
        prefs.edit().putString("theme_design", design.name).apply()
        _uiState.update { it.copy(themeDesign = design) }
    }

    fun setFontSize(fontSize: AppFontSize) {
        prefs.edit().putString("font_size", fontSize.name).apply()
        _uiState.update { it.copy(fontSize = fontSize) }
    }

    fun setSoundEnabled(enabled: Boolean) {
        prefs.edit().putBoolean("sound_enabled", enabled).apply()
        soundManager.isEnabled = enabled
        _uiState.update { it.copy(soundEnabled = enabled) }
    }

    fun navigateToMode(mode: ToolScreenMode) {
        val actions = getAvailableActions(mode = mode, brand = _uiState.value.selectedBrand)
        _uiState.update { state ->
            val nextAction = if (actions.isNotEmpty() && state.selectedAction !in actions) {
                actions.first()
            } else {
                state.selectedAction
            }
            state.copy(
                screenMode = mode,
                selectedAction = nextAction
            )
        }
    }

    fun setBrand(brand: ServiceBrand) {
        val available = getAvailableActions(mode = _uiState.value.screenMode, brand = brand)
        _uiState.update { state ->
            state.copy(
                selectedBrand = brand,
                selectedAction = available.firstOrNull() ?: state.selectedAction
            )
        }
    }

    fun setSelectedAction(action: ServiceActionItem) {
        _uiState.update {
            it.copy(selectedAction = action)
        }
    }

    fun setPartition(partitionLabel: String) {
        val cleanPart = partitionLabel.substringBefore(" ").lowercase(Locale.US)
        _uiState.update { state ->
            val updatedFile = state.selectedFlashFile?.let { file ->
                if (file.steps.size == 1 && cleanPart != "auto") {
                    file.copy(
                        steps = file.steps.map {
                            it.copy(
                                targetPartition = cleanPart,
                                command = "fastboot flash $cleanPart ${file.fileName}"
                            )
                        }
                    )
                } else {
                    file
                }
            }
            state.copy(
                selectedPartition = partitionLabel,
                selectedFlashFile = updatedFile
            )
        }
    }

    fun toggleDisableVerity() {
        _uiState.update { it.copy(disableVerity = !it.disableVerity) }
    }

    fun toggleClearUserdata() {
        _uiState.update { it.copy(clearUserdata = !it.clearUserdata) }
    }

    fun toggleAutoReboot() {
        _uiState.update { it.copy(autoReboot = !it.autoReboot) }
    }

    fun toggleChecksum() {
        _uiState.update { it.copy(checksumVerify = !it.checksumVerify) }
    }

    fun toggleUseFastbootd() {
        _uiState.update { it.copy(useFastbootd = !it.useFastbootd) }
    }

    fun toggleFlashStep(stepId: String) {
        _uiState.update { state ->
            val file = state.selectedFlashFile ?: return@update state
            val updated = file.steps.map { s ->
                if (s.id == stepId) {
                    val nextEnabled = !s.enabled
                    s.copy(
                        enabled = nextEnabled,
                        status = if (nextEnabled) FlashStepStatus.READY else FlashStepStatus.SKIPPED,
                        progress = 0f
                    )
                } else {
                    s
                }
            }
            state.copy(selectedFlashFile = file.copy(steps = updated))
        }
    }

    fun toggleAllFlashSteps(selectAll: Boolean) {
        _uiState.update { state ->
            val file = state.selectedFlashFile ?: return@update state
            val hasExplicitFlashable = file.steps.any { it.isFlashable }
            val updated = file.steps.map { s ->
                val enableTarget = if (selectAll) {
                    if (hasExplicitFlashable) s.isFlashable else true
                } else {
                    false
                }
                s.copy(
                    enabled = enableTarget,
                    status = if (enableTarget) FlashStepStatus.READY else FlashStepStatus.SKIPPED,
                    progress = 0f
                )
            }
            state.copy(selectedFlashFile = file.copy(steps = updated))
        }
    }

    fun scanUsbPorts(userInitiated: Boolean = true) {
        viewModelScope.launch {
            val ports = withContext(Dispatchers.IO) {
                usbController.scanPorts()
            }
            val currentMap = ports.associateBy { it.deviceName }
            val newlyAttached = ports.filter { !knownPortsByName.containsKey(it.deviceName) }
            val detached = knownPortsByName.values.filter { !currentMap.containsKey(it.deviceName) }

            if (hasCompletedInitialUsbScan) {
                if (newlyAttached.isNotEmpty()) {
                    soundManager.playPortAttached()
                } else if (detached.isNotEmpty()) {
                    soundManager.playPortDetached()
                }
            }

            detached.forEach { removed ->
                appendLog(
                    tag = "USB",
                    message = "Port Disconnected: ${removed.productName}",
                    level = LogLevel.WARN
                )
            }

            newlyAttached.forEach { p ->
                appendLog(
                    tag = "USB",
                    message = "Port Connected: ${p.productName} [${p.protocolType.label}]",
                    level = LogLevel.USB
                )
            }

            ports.filter { !it.hasPermission }.forEach { p ->
                usbController.requestPermission(p.deviceName)
            }

            knownPortsByName = currentMap
            hasCompletedInitialUsbScan = true

            val first = ports.firstOrNull()
            _uiState.update {
                it.copy(
                    usbPorts = ports,
                    selectedPortName = first?.deviceName,
                    statusBarText = if (first != null) {
                        "Connected: ${first.productName} (${first.protocolType.label})"
                    } else {
                        "USB OTG Ready"
                    }
                )
            }

            if (userInitiated && ports.isEmpty() && detached.isEmpty()) {
                appendLog(
                    tag = "USB",
                    message = "No USB OTG device detected.",
                    level = LogLevel.INFO
                )
            }
        }
    }

    fun requestActivePortPermission() {
        val port = _uiState.value.usbPorts.firstOrNull()
        if (port != null) {
            usbController.requestPermission(port.deviceName)
        } else {
            scanUsbPorts(userInitiated = true)
        }
    }

    fun onFilePicked(uri: Uri) {
        viewModelScope.launch {
            _uiState.update {
                it.copy(
                    isBusy = true,
                    statusBarText = "Parsing firmware..."
                )
            }
            val context = getApplication<Application>().applicationContext
            val rawPart = _uiState.value.selectedPartition.substringBefore(" ").lowercase(Locale.US)
            val fallbackPart = if (rawPart == "auto") "boot" else rawPart

            val parsed = withContext(Dispatchers.IO) {
                FlashFileProcessor.inspectAndParseUri(context, uri, fallbackPart)
            }
            applyParsedFirmware(parsed)
        }
    }

    fun onFolderPicked(treeUri: Uri) {
        viewModelScope.launch {
            _uiState.update {
                it.copy(
                    isBusy = true,
                    statusBarText = "Scanning firmware folder..."
                )
            }
            val context = getApplication<Application>().applicationContext
            val rawPart = _uiState.value.selectedPartition.substringBefore(" ").lowercase(Locale.US)
            val fallbackPart = if (rawPart == "auto") "boot" else rawPart

            val parsed = withContext(Dispatchers.IO) {
                FlashFileProcessor.inspectAndParseFolderTree(context, treeUri, fallbackPart)
            }
            if (parsed != null) {
                applyParsedFirmware(parsed)
            } else {
                _uiState.update {
                    it.copy(
                        isBusy = false,
                        statusBarText = "No firmware files found in folder"
                    )
                }
                appendLog(
                    tag = "FW",
                    message = "No scatter.txt, rawprogram*.xml, flash.bat, UPDATE.APP, or .img found in folder.",
                    level = LogLevel.WARN
                )
            }
        }
    }

    fun applyParsedFirmware(parsed: SelectedFlashFile) {
        val enabledCount = parsed.steps.count { it.enabled }
        _uiState.update {
            it.copy(
                selectedFlashFile = parsed,
                isBusy = false,
                progressPercent = 0f,
                currentFlashingPartition = "",
                currentChunkLabel = "",
                statusBarText = "${parsed.firmwareInfo.formatLabel}: ${parsed.fileName} ($enabledCount/${parsed.steps.size})",
                screenMode = ToolScreenMode.FASTBOOT_FLASH
            )
        }

        val info = parsed.firmwareInfo
        appendLog(
            tag = "FW",
            message = "FW Format: ${info.formatLabel} | Package: ${parsed.fileName}",
            level = LogLevel.OKAY
        )
        appendLog(
            tag = "FW",
            message = "Platform: ${info.chipsetOrPlatform} | Target: ${info.projectOrBoard} | Storage: ${info.storageType}",
            level = LogLevel.INFO
        )
        appendLog(
            tag = "FW",
            message = "Partitions: $enabledCount/${parsed.steps.size} selected | Total Size: ${parsed.formattedSize} | Sector: ${info.sectorSizeBytes}B",
            level = LogLevel.INFO
        )
        info.extraMetadataLines.forEach { extra ->
            appendLog(tag = "FW", message = extra, level = LogLevel.INFO)
        }
        if (info.largePartitionNames.isNotEmpty()) {
            appendLog(
                tag = "SPLIT",
                message = "Large Partitions (${info.largePartitionNames.joinToString(", ")}) -> Auto Sparse/Chunk Split Ready (128MB/chunk).",
                level = LogLevel.USB
            )
        }
        if (info.missingImageCount > 0 && !info.isFolderPackage) {
            appendLog(
                tag = "FW",
                message = "Manifest parsed (${info.missingImageCount} image payloads external). Load Firmware Folder to attach .img files.",
                level = LogLevel.WARN
            )
        }
    }

    fun onStartPressed() {
        if (_uiState.value.isBusy) {
            stopActiveOperation()
            return
        }
        val state = _uiState.value
        when (state.screenMode) {
            ToolScreenMode.FASTBOOT_FLASH -> executeDirectUsbFlash()
            ToolScreenMode.CMD_TERMINAL -> submitTerminalCommand(state.terminalInput)
            ToolScreenMode.ADB_INFO,
            ToolScreenMode.ADB_UNLOCK,
            ToolScreenMode.ADB_SERVICE,
            ToolScreenMode.FASTBOOT_INFO,
            ToolScreenMode.FASTBOOT_SERVICE,
            ToolScreenMode.FASTBOOT_UNLOCK -> {
                executeServiceAction(state.selectedAction, state.selectedBrand)
            }
            ToolScreenMode.SETTINGS,
            ToolScreenMode.ABOUT -> Unit
        }
    }

    fun stopActiveOperation() {
        val job = activeOperationJob
        activeOperationJob = null
        job?.cancel()
        if (_uiState.value.isBusy) {
            soundManager.playActionStop()
            _uiState.update {
                it.copy(
                    isBusy = false,
                    statusBarText = "Operation Stopped"
                )
            }
            appendLog(
                tag = "STOP",
                message = "Operation stopped.",
                level = LogLevel.WARN
            )
        }
    }

    fun updateTerminalInput(input: String) {
        _uiState.update { it.copy(terminalInput = input) }
    }

    /**
     * Executes pure Termux-style CLI commands strictly inside the dedicated Terminal screen.
     */
    fun submitTerminalCommand(rawCmd: String) {
        if (_uiState.value.isBusy) return
        val cmd = rawCmd.trim()
        _uiState.update { it.copy(terminalInput = "") }

        if (cmd.isEmpty()) {
            appendTerminalLine("~ $ ", isCommand = true)
            return
        }

        if (cmd.equals("clear", ignoreCase = true) || cmd.equals("cls", ignoreCase = true)) {
            _uiState.update { it.copy(terminalLines = emptyList()) }
            return
        }

        appendTerminalLine("~ $ $cmd", isCommand = true)
        _uiState.update { state ->
            val updatedHistory = if (state.terminalHistory.lastOrNull() == cmd) {
                state.terminalHistory
            } else {
                (state.terminalHistory + cmd).takeLast(50)
            }
            state.copy(terminalHistory = updatedHistory)
        }

        if (cmd.equals("help", ignoreCase = true)) {
            appendTerminalLine("fastboot devices | fastboot <cmd> | adb devices | adb shell <cmd> | adb reboot | clear")
            return
        }

        activeOperationJob = viewModelScope.launch {
            _uiState.update { it.copy(isBusy = true) }

            try {
                withContext(Dispatchers.IO) {
                    val lower = cmd.lowercase(Locale.US)
                    val ports = usbController.scanPorts()
                    val activePort = ports.firstOrNull {
                        if (lower.startsWith("fastboot")) {
                            it.protocolType == UsbProtocolType.FASTBOOT
                        } else if (lower.startsWith("adb")) {
                            it.protocolType == UsbProtocolType.ADB
                        } else {
                            true
                        }
                    } ?: ports.firstOrNull()

                    if (activePort != null) {
                        _uiState.update { it.copy(usbPorts = ports, selectedPortName = activePort.deviceName) }
                    }

                    when {
                        lower == "fastboot devices" || lower == "fastboot devices -l" -> {
                            val fbPorts = ports.filter { it.protocolType == UsbProtocolType.FASTBOOT }
                            if (fbPorts.isEmpty()) {
                                ports.forEach { p ->
                                    appendTerminalLine("${p.serialNumber}\t${p.protocolType.label}")
                                }
                            } else {
                                fbPorts.forEach { p ->
                                    appendTerminalLine("${p.serialNumber}\tfastboot")
                                }
                            }
                        }

                        lower == "adb devices" || lower == "adb devices -l" -> {
                            appendTerminalLine("List of devices attached")
                            val adbPorts = ports.filter { it.protocolType == UsbProtocolType.ADB }
                            adbPorts.forEach { p ->
                                val status = if (p.hasPermission) "device" else "unauthorized"
                                appendTerminalLine("${p.serialNumber}\t$status")
                                if (!p.hasPermission) {
                                    usbController.requestPermission(p.deviceName)
                                }
                            }
                        }

                        lower.startsWith("fastboot ") -> {
                            val portName = activePort?.deviceName
                            if (portName == null) {
                                appendTerminalLine("< waiting for any device >", isError = true)
                            } else {
                                val tokens = cmd.split(Regex("\\s+"))
                                val flashIdx = tokens.indexOfFirst { it.equals("flash", ignoreCase = true) }
                                val loadedFile = _uiState.value.selectedFlashFile
                                if (flashIdx >= 0 && flashIdx + 1 < tokens.size && loadedFile != null) {
                                    val targetPart = tokens[flashIdx + 1]
                                    val uri = Uri.parse(loadedFile.uriString)
                                    usbController.flashBinaryStreamOverUsb(
                                        deviceName = portName,
                                        uri = uri,
                                        partition = targetPart,
                                        dataOffset = 0L,
                                        dataSize = loadedFile.fileSizeBytes,
                                        disableVerity = _uiState.value.disableVerity,
                                        verboseWireLogs = false,
                                        onProgress = {},
                                        onLog = { msg, lvl ->
                                            appendTerminalLine(msg, isError = lvl == LogLevel.FAIL)
                                        }
                                    )
                                } else {
                                    val wireList = toFastbootWireProtocolList(cmd)
                                    usbController.executeFastbootWireCommands(
                                        deviceName = portName,
                                        wireCommands = wireList,
                                        verboseWireLogs = false,
                                        onLog = { msg, lvl ->
                                            appendTerminalLine(msg, isError = lvl == LogLevel.FAIL)
                                        }
                                    )
                                }
                            }
                        }

                        lower == "adb" || lower.startsWith("adb ") -> {
                            val portName = activePort?.deviceName
                            if (portName == null) {
                                appendTerminalLine("error: no devices/emulators found", isError = true)
                            } else {
                                usbController.executeAdbOverUsbOtg(
                                    deviceName = portName,
                                    command = cmd,
                                    verboseWireLogs = false,
                                    onLog = { msg, lvl ->
                                        appendTerminalLine(msg, isError = lvl == LogLevel.FAIL)
                                    }
                                )
                            }
                        }

                        else -> {
                            val portName = activePort?.deviceName
                            if (portName != null && activePort.protocolType == UsbProtocolType.ADB) {
                                usbController.executeAdbOverUsbOtg(
                                    deviceName = portName,
                                    command = cmd,
                                    verboseWireLogs = false,
                                    onLog = { msg, lvl ->
                                        appendTerminalLine(msg, isError = lvl == LogLevel.FAIL)
                                    }
                                )
                            } else {
                                usbController.executeLocalShell(
                                    command = cmd,
                                    onLog = { msg, lvl ->
                                        appendTerminalLine(msg, isError = lvl == LogLevel.FAIL)
                                    }
                                )
                            }
                        }
                    }
                }
            } finally {
                _uiState.update { it.copy(isBusy = false) }
            }
        }
    }

    private fun appendTerminalLine(text: String, isCommand: Boolean = false, isError: Boolean = false) {
        val entry = TerminalLine(
            id = terminalSequence.getAndIncrement(),
            text = text,
            isCommand = isCommand,
            isError = isError
        )
        _uiState.update { s ->
            s.copy(terminalLines = (s.terminalLines + entry).takeLast(500))
        }
    }

    fun executeDirectUsbFlash() {
        if (_uiState.value.isBusy) return
        val state = _uiState.value
        val file = state.selectedFlashFile ?: run {
            appendLog("FLASH", "Load a firmware file or folder first.", LogLevel.WARN)
            return
        }
        val enabledSteps = file.steps.filter { it.enabled }
        if (enabledSteps.isEmpty()) {
            appendLog("FLASH", "No partitions selected in Partition Table.", LogLevel.WARN)
            return
        }
        val portName = state.selectedPortName ?: run {
            appendLog("FLASH", "No USB OTG device connected.", LogLevel.FAIL)
            return
        }

        // Reset step states cleanly before starting
        updateFlashSteps { step ->
            if (step.enabled) {
                step.copy(status = FlashStepStatus.QUEUED, progress = 0f)
            } else {
                step.copy(status = FlashStepStatus.SKIPPED, progress = 0f)
            }
        }

        soundManager.playActionStart()
        activeOperationJob = viewModelScope.launch {
            var completedNormally = false
            try {
                _uiState.update {
                    it.copy(
                        isBusy = true,
                        progressPercent = 0f,
                        currentFlashingPartition = "",
                        currentChunkLabel = "",
                        statusBarText = "Starting Flash (${enabledSteps.size} partitions)..."
                    )
                }
                appendLog(
                    tag = "FLASH",
                    message = "Flash Start: ${file.fileName} | Selected: ${enabledSteps.size} | ClearUserdata: ${if (state.clearUserdata) "ON" else "OFF"} | AutoReboot: ${if (state.autoReboot) "ON" else "OFF"}",
                    level = LogLevel.CMD
                )

                val totalWeightBytes = enabledSteps.sumOf { it.dataSize.coerceAtLeast(1024L * 1024L) }
                    .coerceAtLeast(1L)
                var finishedWeightBytes = 0L
                var allSucceeded = true

                withContext(Dispatchers.IO) {
                    if (state.useFastbootd) {
                        appendLog("FLASH", "Switching to Fastbootd...", LogLevel.USB)
                        usbController.executeFastbootWireCommands(
                            deviceName = portName,
                            wireCommands = listOf("reboot-fastboot"),
                            verboseWireLogs = false
                        ) { msg, lvl ->
                            appendLog("FLASH", msg, lvl)
                        }
                    }

                    for (idx in enabledSteps.indices) {
                        coroutineContext.ensureActive()
                        val step = enabledSteps[idx]
                        val stepWeight = step.dataSize.coerceAtLeast(1024L * 1024L)
                        var currentChunkSuffix = ""
                        var lastLoggedPctMilestone = 0

                        updateSingleFlashStep(step.id, FlashStepStatus.FLASHING, 0f)
                        _uiState.update {
                            it.copy(
                                currentFlashingPartition = step.targetPartition,
                                currentChunkLabel = "",
                                statusBarText = "[${idx + 1}/${enabledSteps.size}] Flashing ${step.targetPartition} (0%)..."
                            )
                        }
                        appendLog(
                            tag = "FLASH",
                            message = "[${idx + 1}/${enabledSteps.size}] Writing ${step.targetPartition} (${FlashFileProcessor.formatSize(step.dataSize)}) — 0%",
                            level = LogLevel.CMD
                        )

                        val resolvedUriStr = step.sourceUriString
                            ?: file.uriString.takeIf {
                                file.fileType == com.example.model.FlashFileType.UPDATE_APP ||
                                    file.fileType == com.example.model.FlashFileType.IMAGE ||
                                    file.fileType == com.example.model.FlashFileType.RAW
                            }

                        val stepOk: Boolean = when {
                            resolvedUriStr != null && step.dataSize > 0L -> {
                                val stepUri = Uri.parse(resolvedUriStr)
                                usbController.flashBinaryStreamOverUsb(
                                    deviceName = portName,
                                    uri = stepUri,
                                    partition = step.targetPartition,
                                    dataOffset = step.dataOffset,
                                    dataSize = step.dataSize,
                                    disableVerity = state.disableVerity,
                                    verboseWireLogs = false,
                                    isCancelled = { activeOperationJob?.isActive != true },
                                    onChunkInfo = { cIdx, cTotal, _ ->
                                        currentChunkSuffix = if (cTotal > 1) "Chunk $cIdx/$cTotal" else ""
                                        _uiState.update { it.copy(currentChunkLabel = currentChunkSuffix) }
                                    },
                                    onProgress = { ratio ->
                                        val clamped = ratio.coerceIn(0f, 1f)
                                        val partPct = (clamped * 100f).toInt()
                                        val globalPct = ((finishedWeightBytes + stepWeight * clamped) / totalWeightBytes.toFloat()) * 100f

                                        updateSingleFlashStep(step.id, FlashStepStatus.FLASHING, clamped)
                                        _uiState.update {
                                            it.copy(
                                                progressPercent = globalPct.coerceIn(0f, 100f),
                                                statusBarText = "[${idx + 1}/${enabledSteps.size}] ${step.targetPartition}: $partPct% ${if (currentChunkSuffix.isNotEmpty()) "• $currentChunkSuffix" else ""}".trim()
                                            )
                                        }

                                        val milestone = (partPct / 25) * 25
                                        if (milestone in listOf(25, 50, 75) && milestone > lastLoggedPctMilestone) {
                                            lastLoggedPctMilestone = milestone
                                            appendLog(
                                                tag = "PROG",
                                                message = "  ↳ ${step.targetPartition}: $milestone% ${if (currentChunkSuffix.isNotEmpty()) "($currentChunkSuffix)" else ""}".trim(),
                                                level = LogLevel.USB
                                            )
                                        }
                                    },
                                    onLog = { msg, lvl -> appendLog("USB", msg, lvl) }
                                )
                            }

                            step.command.startsWith("fastboot flash", ignoreCase = true) -> {
                                appendLog(
                                    tag = "SKIP",
                                    message = "  ↳ Payload missing for ${step.targetPartition} (${step.sourceName}). Load Firmware Folder.",
                                    level = LogLevel.WARN
                                )
                                false
                            }

                            else -> {
                                val wireCmds = toFastbootWireProtocolList(step.command)
                                usbController.executeFastbootWireCommands(
                                    deviceName = portName,
                                    wireCommands = wireCmds,
                                    verboseWireLogs = false,
                                    onLog = { msg, lvl -> appendLog("USB", msg, lvl) }
                                )
                            }
                        }

                        finishedWeightBytes += stepWeight
                        if (stepOk) {
                            updateSingleFlashStep(step.id, FlashStepStatus.DONE, 1f)
                            appendLog(
                                tag = "OKAY",
                                message = "[${idx + 1}/${enabledSteps.size}] ${step.targetPartition} — 100% DONE",
                                level = LogLevel.OKAY
                            )
                        } else {
                            allSucceeded = false
                            updateSingleFlashStep(step.id, FlashStepStatus.ERROR, 0f)
                            appendLog(
                                tag = "FAIL",
                                message = "[${idx + 1}/${enabledSteps.size}] ${step.targetPartition} — FAILED",
                                level = LogLevel.FAIL
                            )
                        }
                    }

                    if (state.clearUserdata) {
                        coroutineContext.ensureActive()
                        _uiState.update { it.copy(statusBarText = "Wiping userdata & metadata...") }
                        appendLog("WIPE", "Clearing userdata & metadata...", LogLevel.CMD)
                        usbController.executeFastbootWireCommands(
                            deviceName = portName,
                            wireCommands = listOf("erase:userdata", "erase:metadata"),
                            verboseWireLogs = false
                        ) { msg, lvl ->
                            appendLog("WIPE", msg, lvl)
                        }
                    }

                    if (state.autoReboot) {
                        coroutineContext.ensureActive()
                        _uiState.update { it.copy(statusBarText = "Rebooting device...") }
                        appendLog("FLASH", "Auto Rebooting device...", LogLevel.INFO)
                        usbController.executeFastbootWireCommands(
                            deviceName = portName,
                            wireCommands = listOf("reboot"),
                            verboseWireLogs = false
                        ) { msg, lvl ->
                            appendLog("USB", msg, lvl)
                        }
                    } else {
                        appendLog("FLASH", "Auto Reboot OFF — staying in Fastboot mode.", LogLevel.INFO)
                    }
                }
                completedNormally = allSucceeded
                appendLog(
                    tag = "DONE",
                    message = if (allSucceeded) "All selected partitions flashed (100%)." else "Flash finished with errors.",
                    level = if (allSucceeded) LogLevel.OKAY else LogLevel.WARN
                )
            } catch (c: CancellationException) {
                throw c
            } finally {
                if (completedNormally) {
                    soundManager.playActionStop()
                }
                _uiState.update {
                    it.copy(
                        isBusy = false,
                        progressPercent = if (completedNormally) 100f else it.progressPercent,
                        currentFlashingPartition = "",
                        currentChunkLabel = "",
                        statusBarText = if (completedNormally) "Flash Completed (100%)" else "USB OTG Ready"
                    )
                }
            }
        }
    }

    private fun updateFlashSteps(transform: (ParsedFlashStep) -> ParsedFlashStep) {
        _uiState.update { state ->
            val file = state.selectedFlashFile ?: return@update state
            state.copy(
                selectedFlashFile = file.copy(
                    steps = file.steps.map(transform)
                )
            )
        }
    }

    private fun updateSingleFlashStep(stepId: String, status: FlashStepStatus, progress: Float) {
        updateFlashSteps { step ->
            if (step.id == stepId) step.copy(status = status, progress = progress) else step
        }
    }

    private fun executeServiceAction(action: ServiceActionItem, brand: ServiceBrand) {
        if (_uiState.value.isBusy) return
        val cmd = action.command.trim()
        if (cmd.isEmpty()) return

        soundManager.playActionStart()
        activeOperationJob = viewModelScope.launch {
            var completedNormally = false
            try {
                _uiState.update {
                    it.copy(
                        isBusy = true,
                        statusBarText = "Running: ${action.title}",
                        progressPercent = 0f
                    )
                }
                appendLog(
                    tag = "START",
                    message = "[${brand.shortTag}] ${action.title}",
                    level = LogLevel.CMD
                )

                val state = _uiState.value
                val portName = state.selectedPortName

                withContext(Dispatchers.IO) {
                    when {
                        cmd.startsWith("fastboot", ignoreCase = true) -> {
                            if (portName == null) {
                                appendLog(
                                    tag = "FAIL",
                                    message = "No Fastboot USB OTG device connected.",
                                    level = LogLevel.FAIL
                                )
                            } else {
                                val wireList = toFastbootWireProtocolList(cmd)
                                usbController.executeFastbootWireCommands(
                                    deviceName = portName,
                                    wireCommands = wireList,
                                    verboseWireLogs = false,
                                    onLog = { msg, lvl -> appendLog("FB", msg, lvl) }
                                )
                            }
                        }
                        else -> {
                            if (portName != null) {
                                usbController.executeAdbOverUsbOtg(
                                    deviceName = portName,
                                    command = cmd,
                                    verboseWireLogs = false,
                                    onLog = { msg, lvl -> appendLog("ADB", msg, lvl) }
                                )
                            } else {
                                appendLog(
                                    tag = "FAIL",
                                    message = "No ADB USB OTG device connected.",
                                    level = LogLevel.FAIL
                                )
                            }
                        }
                    }
                }
                completedNormally = true
                appendLog(
                    tag = "DONE",
                    message = "${action.title} finished.",
                    level = LogLevel.OKAY
                )
            } catch (c: CancellationException) {
                throw c
            } finally {
                if (completedNormally) {
                    soundManager.playActionStop()
                }
                _uiState.update {
                    it.copy(
                        isBusy = false,
                        statusBarText = "USB OTG Ready"
                    )
                }
            }
        }
    }

    private fun toFastbootWireProtocolList(cliCommand: String): List<String> {
        val body = cliCommand
            .replace(Regex("^fastboot(\\s+--[a-zA-Z0-9_-]+)*\\s+", RegexOption.IGNORE_CASE), "")
            .trim()
        return when {
            body.equals("-w", ignoreCase = true) -> listOf("erase:userdata", "erase:metadata")
            body.equals("erase modemst1", ignoreCase = true) -> listOf("erase:modemst1", "erase:modemst2")
            body.equals("reboot fastboot", ignoreCase = true) -> listOf("reboot-fastboot")
            body.equals("reboot recovery", ignoreCase = true) -> listOf("reboot-recovery")
            body.startsWith("getvar ", ignoreCase = true) -> listOf("getvar:${body.substringAfter(" ").trim()}")
            body.startsWith("erase ", ignoreCase = true) -> listOf("erase:${body.substringAfter(" ").trim()}")
            body.startsWith("set_active ", ignoreCase = true) -> listOf("set_active:${body.substringAfter(" ").trim()}")
            body.startsWith("--set-active=", ignoreCase = true) -> listOf("set_active:${body.substringAfter("=").trim()}")
            body.startsWith("snapshot-update ", ignoreCase = true) -> listOf("snapshot-update:${body.substringAfter(" ").trim()}")
            body.startsWith("create-logical-partition ", ignoreCase = true) -> {
                val args = body.removePrefix("create-logical-partition ").trim().replace(Regex("\\s+"), ":")
                listOf("create-logical-partition:$args")
            }
            body.startsWith("delete-logical-partition ", ignoreCase = true) ->
                listOf("delete-logical-partition:${body.substringAfter(" ").trim()}")
            body.startsWith("resize-logical-partition ", ignoreCase = true) -> {
                val args = body.removePrefix("resize-logical-partition ").trim().replace(Regex("\\s+"), ":")
                listOf("resize-logical-partition:$args")
            }
            else -> listOf(body)
        }
    }

    fun clearLogs() {
        _uiState.update { it.copy(consoleLogs = emptyList()) }
    }

    private fun appendLog(tag: String, message: String, level: LogLevel) {
        val ts = SimpleDateFormat("HH:mm:ss.SSS", Locale.US).format(Date())
        val entry = ConsoleLogEntry(
            id = logSequence.getAndIncrement(),
            timestamp = ts,
            tag = tag,
            message = message,
            level = level
        )
        _uiState.update { s ->
            s.copy(consoleLogs = (s.consoleLogs + entry).takeLast(300))
        }
    }
}

/**
 * Pure Termux-style Fullscreen CLI Terminal Box.
 * Relies cleanly on parent Scaffold WindowInsets.safeDrawing padding (no duplicate imePadding).
 * Inherits active ThemeDesign (Cyber Neon, Emerald Matrix, Crimson Forge), Dark/Light mode, and FontSize.
 */
@Composable
fun FullscreenCliTerminalScreen(
    uiState: ServiceUiState,
    onTerminalInputChange: (String) -> Unit,
    onSubmitCommand: (String) -> Unit,
    modifier: Modifier = Modifier
) {
    val palette = LocalAppPalette.current
    val fontConfig = LocalFontConfig.current

    val terminalBg = palette.terminalBg
    val dockBg = palette.headerBg
    val dividerColor = palette.divider
    val promptColor = palette.accentPrimary
    val inputColor = palette.textPrimary
    val hintColor = palette.textMuted
    val keyBg = palette.cardAltBg
    val keyBorder = palette.cardBorder
    val keyTextColor = palette.textPrimary

    val listState = rememberLazyListState()
    val focusRequester = remember { FocusRequester() }
    val keyboardController = LocalSoftwareKeyboardController.current
    val interactionSource = remember { MutableInteractionSource() }

    var textFieldValue by remember {
        mutableStateOf(
            TextFieldValue(
                text = uiState.terminalInput,
                selection = TextRange(uiState.terminalInput.length)
            )
        )
    }
    var historyCursor by remember { mutableIntStateOf(-1) }

    LaunchedEffect(uiState.terminalInput) {
        if (uiState.terminalInput != textFieldValue.text) {
            textFieldValue = TextFieldValue(
                text = uiState.terminalInput,
                selection = TextRange(uiState.terminalInput.length)
            )
        }
    }

    LaunchedEffect(uiState.terminalLines.size) {
        if (uiState.terminalLines.isNotEmpty()) {
            listState.animateScrollToItem(uiState.terminalLines.lastIndex)
        }
    }

    fun submitCurrent() {
        if (uiState.isBusy) return
        historyCursor = -1
        onSubmitCommand(textFieldValue.text)
    }

    fun insertToken(token: String) {
        val current = textFieldValue.text
        val nextText = when {
            current.isBlank() -> "$token "
            current.endsWith(" ") -> "$current$token "
            else -> "$current $token "
        }
        historyCursor = -1
        textFieldValue = TextFieldValue(
            text = nextText,
            selection = TextRange(nextText.length)
        )
        onTerminalInputChange(nextText)
        try {
            focusRequester.requestFocus()
            keyboardController?.show()
        } catch (_: Exception) {
        }
    }

    fun stepHistoryUp() {
        val history = uiState.terminalHistory
        if (history.isEmpty()) return
        val nextCursor = if (historyCursor == -1) {
            history.lastIndex
        } else {
            (historyCursor - 1).coerceAtLeast(0)
        }
        historyCursor = nextCursor
        val cmd = history[nextCursor]
        textFieldValue = TextFieldValue(text = cmd, selection = TextRange(cmd.length))
        onTerminalInputChange(cmd)
    }

    fun stepHistoryDown() {
        val history = uiState.terminalHistory
        if (history.isEmpty() || historyCursor == -1) return
        val nextCursor = historyCursor + 1
        if (nextCursor > history.lastIndex) {
            historyCursor = -1
            textFieldValue = TextFieldValue(text = "", selection = TextRange(0))
            onTerminalInputChange("")
        } else {
            historyCursor = nextCursor
            val cmd = history[nextCursor]
            textFieldValue = TextFieldValue(text = cmd, selection = TextRange(cmd.length))
            onTerminalInputChange(cmd)
        }
    }

    Column(
        modifier = modifier
            .fillMaxSize()
            .background(terminalBg)
            .testTag("fullscreen_termux_terminal")
    ) {
        // Scrollable, copy-selectable CLI output area (never wraps BasicTextField)
        Box(
            modifier = Modifier
                .weight(1f)
                .fillMaxWidth()
                .clickable(
                    interactionSource = interactionSource,
                    indication = null
                ) {
                    try {
                        focusRequester.requestFocus()
                        keyboardController?.show()
                    } catch (_: Exception) {
                    }
                }
        ) {
            if (uiState.terminalLines.isEmpty()) {
                Text(
                    text = "~ $ ready (type 'help', 'adb devices', or 'fastboot devices')",
                    fontFamily = FontFamily.Monospace,
                    fontSize = fontConfig.terminalOutSp,
                    color = hintColor,
                    modifier = Modifier.padding(horizontal = 12.dp, vertical = 12.dp)
                )
            } else {
                SelectionContainer {
                    LazyColumn(
                        state = listState,
                        contentPadding = PaddingValues(horizontal = 10.dp, vertical = 10.dp),
                        modifier = Modifier.fillMaxSize()
                    ) {
                        items(uiState.terminalLines, key = { it.id }) { line ->
                            TermuxOutputLine(line = line)
                        }
                    }
                }
            }
        }

        // Termux Extra-Keys Bar
        HorizontalDivider(color = dividerColor)
        Row(
            modifier = Modifier
                .fillMaxWidth()
                .background(dockBg)
                .horizontalScroll(rememberScrollState())
                .padding(horizontal = 8.dp, vertical = 5.dp),
            horizontalArrangement = Arrangement.spacedBy(6.dp),
            verticalAlignment = Alignment.CenterVertically
        ) {
            TermuxExtraKey("adb", keyBg, keyBorder, keyTextColor) { insertToken("adb") }
            TermuxExtraKey("fastboot", keyBg, keyBorder, keyTextColor) { insertToken("fastboot") }
            TermuxExtraKey("shell", keyBg, keyBorder, keyTextColor) { insertToken("shell") }
            TermuxExtraKey("devices", keyBg, keyBorder, keyTextColor) { insertToken("devices") }
            TermuxExtraKey("reboot", keyBg, keyBorder, keyTextColor) { insertToken("reboot") }
            TermuxExtraKey("↑", keyBg, keyBorder, promptColor) { stepHistoryUp() }
            TermuxExtraKey("↓", keyBg, keyBorder, promptColor) { stepHistoryDown() }
            TermuxExtraKey("CLS", keyBg, keyBorder, palette.accentSecondary) {
                historyCursor = -1
                onSubmitCommand("clear")
            }
        }
        HorizontalDivider(color = dividerColor)

        // Pinned Termux CLI Input Prompt Row
        Row(
            modifier = Modifier
                .fillMaxWidth()
                .background(dockBg)
                .padding(start = 10.dp, end = 4.dp, top = 4.dp, bottom = 4.dp),
            verticalAlignment = Alignment.CenterVertically
        ) {
            Text(
                text = "~ $ ",
                fontFamily = FontFamily.Monospace,
                fontSize = fontConfig.terminalCmdSp,
                fontWeight = FontWeight.Bold,
                color = promptColor
            )
            Box(
                contentAlignment = Alignment.CenterStart,
                modifier = Modifier
                    .weight(1f)
                    .defaultMinSize(minHeight = 40.dp)
            ) {
                if (uiState.isBusy) {
                    Text(
                        text = "running...",
                        fontFamily = FontFamily.Monospace,
                        fontSize = fontConfig.terminalCmdSp,
                        color = hintColor
                    )
                } else {
                    if (textFieldValue.text.isEmpty()) {
                        Text(
                            text = "adb / fastboot command...",
                            fontFamily = FontFamily.Monospace,
                            fontSize = fontConfig.terminalOutSp,
                            color = hintColor
                        )
                    }
                    BasicTextField(
                        value = textFieldValue,
                        onValueChange = { next ->
                            if (next.text.contains('\n')) {
                                val clean = next.text.replace("\n", "")
                                historyCursor = -1
                                textFieldValue = TextFieldValue("", TextRange(0))
                                onSubmitCommand(clean)
                            } else {
                                textFieldValue = next
                                onTerminalInputChange(next.text)
                            }
                        },
                        singleLine = true,
                        cursorBrush = SolidColor(promptColor),
                        textStyle = TextStyle(
                            color = inputColor,
                            fontFamily = FontFamily.Monospace,
                            fontSize = fontConfig.terminalCmdSp,
                            lineHeight = 20.sp
                        ),
                        keyboardOptions = KeyboardOptions(
                            keyboardType = KeyboardType.Ascii,
                            autoCorrectEnabled = false,
                            imeAction = ImeAction.Send
                        ),
                        keyboardActions = KeyboardActions(
                            onSend = { submitCurrent() },
                            onDone = { submitCurrent() }
                        ),
                        modifier = Modifier
                            .fillMaxWidth()
                            .focusRequester(focusRequester)
                            .onPreviewKeyEvent { keyEvent ->
                                if (keyEvent.type == KeyEventType.KeyDown) {
                                    when (keyEvent.key) {
                                        Key.Enter, Key.NumPadEnter -> {
                                            submitCurrent()
                                            true
                                        }
                                        Key.DirectionUp -> {
                                            stepHistoryUp()
                                            true
                                        }
                                        Key.DirectionDown -> {
                                            stepHistoryDown()
                                            true
                                        }
                                        else -> false
                                    }
                                } else {
                                    false
                                }
                            }
                            .testTag("termux_cli_input")
                    )
                }
            }
            Spacer(modifier = Modifier.width(2.dp))
            IconButton(
                onClick = { submitCurrent() },
                enabled = !uiState.isBusy,
                modifier = Modifier
                    .size(48.dp)
                    .testTag("termux_send_button")
            ) {
                Icon(
                    imageVector = Icons.AutoMirrored.Filled.Send,
                    contentDescription = "Run Command",
                    tint = if (uiState.isBusy) hintColor else promptColor,
                    modifier = Modifier.size(20.dp)
                )
            }
        }
    }
}

@Composable
private fun TermuxOutputLine(
    line: TerminalLine
) {
    val palette = LocalAppPalette.current
    val fontConfig = LocalFontConfig.current

    val promptColor = palette.accentPrimary
    val cmdColor = palette.textPrimary
    val outputColor = if (palette.isDark) Color(0xFFE5E7EB) else palette.textSecondary
    val errorColor = if (palette.isDark) Color(0xFFF87171) else Color(0xFFDC2626)

    if (line.isCommand) {
        val promptPrefix = "~ $ "
        val cmdBody = line.text.removePrefix(promptPrefix)
        Text(
            text = buildAnnotatedString {
                withStyle(SpanStyle(color = promptColor, fontWeight = FontWeight.Bold)) {
                    append(promptPrefix)
                }
                withStyle(SpanStyle(color = cmdColor)) {
                    append(cmdBody)
                }
            },
            fontFamily = FontFamily.Monospace,
            fontSize = fontConfig.terminalCmdSp,
            lineHeight = 19.sp,
            modifier = Modifier
                .fillMaxWidth()
                .padding(vertical = 1.dp)
        )
    } else {
        Text(
            text = line.text,
            fontFamily = FontFamily.Monospace,
            fontSize = fontConfig.terminalOutSp,
            lineHeight = 18.sp,
            color = if (line.isError) errorColor else outputColor,
            modifier = Modifier
                .fillMaxWidth()
                .padding(vertical = 1.dp)
        )
    }
}

@Composable
private fun PartitionTableRow(
    step: ParsedFlashStep,
    isBusy: Boolean,
    onToggle: () -> Unit
) {
    val palette = LocalAppPalette.current
    val isLarge = step.dataSize > SparseChunkPlanner.DEFAULT_SAFE_CHUNK_BYTES

    val statusColor = when (step.status) {
        FlashStepStatus.READY -> palette.accentPrimary
        FlashStepStatus.QUEUED -> palette.textSecondary
        FlashStepStatus.FLASHING -> Color(0xFFF59E0B)
        FlashStepStatus.DONE -> Color(0xFF10B981)
        FlashStepStatus.SKIPPED -> palette.textMuted
        FlashStepStatus.ERROR -> Color(0xFFEF4444)
    }

    val statusText = when (step.status) {
        FlashStepStatus.FLASHING -> "${(step.progress * 100).toInt()}%"
        else -> step.status.label
    }

    Column(
        modifier = Modifier
            .fillMaxWidth()
            .background(
                if (step.status == FlashStepStatus.FLASHING) {
                    palette.accentPrimary.copy(alpha = 0.08f)
                } else {
                    Color.Transparent
                }
            )
            .clickable(enabled = !isBusy) { onToggle() }
            .padding(horizontal = 10.dp, vertical = 6.dp)
    ) {
        Row(
            modifier = Modifier.fillMaxWidth(),
            verticalAlignment = Alignment.CenterVertically
        ) {
            Icon(
                imageVector = if (step.enabled) Icons.Default.CheckBox else Icons.Default.CheckBoxOutlineBlank,
                contentDescription = "Toggle ${step.targetPartition}",
                tint = if (step.enabled) palette.accentPrimary else palette.textMuted,
                modifier = Modifier.size(18.dp)
            )
            Spacer(modifier = Modifier.width(8.dp))

            // Partition Name + Tags
            Column(modifier = Modifier.weight(1.1f)) {
                Text(
                    text = step.targetPartition,
                    style = MaterialTheme.typography.labelSmall,
                    fontFamily = FontFamily.Monospace,
                    fontWeight = FontWeight.Bold,
                    color = if (step.enabled) palette.textPrimary else palette.textMuted,
                    maxLines = 1,
                    overflow = TextOverflow.Ellipsis
                )
                Row(horizontalArrangement = Arrangement.spacedBy(3.dp)) {
                    if (step.isSparse) {
                        MiniBadge("SPARSE", palette.accentSecondary)
                    }
                    if (isLarge) {
                        MiniBadge("SPLIT", Color(0xFFF59E0B))
                    }
                }
            }

            // Size
            Text(
                text = if (step.dataSize > 0L) FlashFileProcessor.formatSize(step.dataSize) else "CMD",
                style = MaterialTheme.typography.labelSmall,
                fontSize = 10.sp,
                fontFamily = FontFamily.Monospace,
                color = palette.textSecondary,
                maxLines = 1,
                modifier = Modifier.weight(0.65f)
            )

            // Source File + Offset/Sector
            Column(modifier = Modifier.weight(1.15f)) {
                Text(
                    text = step.sourceName,
                    style = MaterialTheme.typography.labelSmall,
                    fontSize = 10.sp,
                    color = if (step.sourceUriString == null && step.isFlashable && step.command.startsWith("fastboot flash")) {
                        Color(0xFFF59E0B)
                    } else {
                        palette.textSecondary
                    },
                    maxLines = 1,
                    overflow = TextOverflow.Ellipsis
                )
                if (step.startAddressHex.isNotEmpty()) {
                    Text(
                        text = step.startAddressHex,
                        style = MaterialTheme.typography.labelSmall,
                        fontSize = 9.sp,
                        fontFamily = FontFamily.Monospace,
                        color = palette.textMuted,
                        maxLines = 1,
                        overflow = TextOverflow.Ellipsis
                    )
                }
            }

            // Status Badge
            Surface(
                shape = RoundedCornerShape(4.dp),
                color = statusColor.copy(alpha = 0.14f),
                border = BorderStroke(1.dp, statusColor.copy(alpha = 0.45f)),
                modifier = Modifier.width(56.dp)
            ) {
                Box(
                    contentAlignment = Alignment.Center,
                    modifier = Modifier.padding(vertical = 2.dp)
                ) {
                    Text(
                        text = statusText,
                        style = MaterialTheme.typography.labelSmall,
                        fontSize = 9.sp,
                        fontFamily = FontFamily.Monospace,
                        fontWeight = FontWeight.Bold,
                        color = statusColor,
                        maxLines = 1
                    )
                }
            }
        }

        if (step.status == FlashStepStatus.FLASHING) {
            Spacer(modifier = Modifier.height(4.dp))
            LinearProgressIndicator(
                progress = { step.progress.coerceIn(0f, 1f) },
                color = palette.accentPrimary,
                trackColor = palette.cardBorder,
                modifier = Modifier
                    .fillMaxWidth()
                    .height(2.dp)
            )
        }
    }
}

@Composable
private fun ConsoleTimestampRow(
    entry: ConsoleLogEntry,
    consoleFontSp: androidx.compose.ui.unit.TextUnit
) {
    val palette = LocalAppPalette.current
    val msgColor = if (palette.isDark) {
        when (entry.level) {
            LogLevel.INIT -> palette.accentPrimary
            LogLevel.CMD -> palette.accentSecondary
            LogLevel.USB -> Color(0xFF60A5FA)
            LogLevel.OKAY -> Color(0xFF34D399)
            LogLevel.INFO -> palette.textPrimary
            LogLevel.WARN -> Color(0xFFFACC15)
            LogLevel.FAIL -> Color(0xFFEF4444)
        }
    } else {
        when (entry.level) {
            LogLevel.INIT -> palette.accentPrimary
            LogLevel.CMD -> palette.accentSecondary
            LogLevel.USB -> Color(0xFF1D4ED8)
            LogLevel.OKAY -> Color(0xFF059669)
            LogLevel.INFO -> palette.textPrimary
            LogLevel.WARN -> Color(0xFFB45309)
            LogLevel.FAIL -> Color(0xFFDC2626)
        }
    }

    Row(
        modifier = Modifier.fillMaxWidth(),
        verticalAlignment = Alignment.Top
    ) {
        Text(
            text = "[${entry.timestamp}]",
            fontSize = consoleFontSp,
            fontFamily = FontFamily.Monospace,
            color = palette.textMuted
        )
        Spacer(modifier = Modifier.width(8.dp))
        Text(
            text = entry.message,
            fontSize = consoleFontSp,
            fontFamily = FontFamily.Monospace,
            color = msgColor,
            modifier = Modifier.weight(1f)
        )
    }
}

private fun buildScaledTypography(scale: Float): Typography {
    return Typography(
        titleLarge = Typography.titleLarge.copy(fontSize = (22f * scale).sp),
        titleMedium = Typography.titleMedium.copy(fontSize = (16f * scale).sp),
        bodyLarge = Typography.bodyLarge.copy(fontSize = (16f * scale).sp),
        bodyMedium = Typography.bodyMedium.copy(fontSize = (14f * scale).sp),
        bodySmall = Typography.bodySmall.copy(fontSize = (12f * scale).sp),
        labelLarge = Typography.labelLarge.copy(fontSize = (14f * scale).sp),
        labelMedium = Typography.labelMedium.copy(fontSize = (12f * scale).sp),
        labelSmall = Typography.labelSmall.copy(fontSize = (11f * scale).sp)
    )

@Composable
fun MyApplicationTheme(
    themeMode: ThemeMode = ThemeMode.DARK,
    themeDesign: ThemeDesign = ThemeDesign.CYBER_NEON,
    fontSize: AppFontSize = AppFontSize.MEDIUM,
    content: @Composable () -> Unit
) {
    val palette = resolvePalette(themeMode, themeDesign)
    val fontConfig = FontConfig(fontSize)

    val colorScheme = if (palette.isDark) {
        darkColorScheme(
            primary = palette.accentPrimary,
            onPrimary = Color.Black,
            secondary = palette.accentSecondary,
            onSecondary = Color.Black,
            background = palette.screenBg,
            onBackground = palette.textPrimary,
            surface = palette.cardBg,
            onSurface = palette.textPrimary,
            surfaceVariant = palette.cardAltBg,
            onSurfaceVariant = palette.textSecondary,
            outline = palette.cardBorder
        )
    } else {
        lightColorScheme(
            primary = palette.accentPrimary,
            onPrimary = Color.White,
            secondary = palette.accentSecondary,
            onSecondary = Color.White,
            background = palette.screenBg,
            onBackground = palette.textPrimary,
            surface = palette.cardBg,
            onSurface = palette.textPrimary,
            surfaceVariant = palette.cardAltBg,
            onSurfaceVariant = palette.textSecondary,
            outline = palette.cardBorder
        )
    }

    CompositionLocalProvider(
        LocalAppPalette provides palette,
        LocalFontConfig provides fontConfig
    ) {
        MaterialTheme(
            colorScheme = colorScheme,
            typography = buildScaledTypography(fontSize.scaleFactor),
            content = content
        )
    }
}

    @Test
    fun `verify ADB RSA public key encodes exact 524-byte struct and signs 20-byte token validly`() {
        val kpg = KeyPairGenerator.getInstance("RSA")
        kpg.initialize(2048)
        val kp = kpg.generateKeyPair()
        val pub = kp.public as RSAPublicKey

        val payload = UsbPortController.encodeAdbPublicKeyPayload(pub)
        assertEquals(0.toByte(), payload.last())
        val payloadStr = String(payload, 0, payload.size - 1, StandardCharsets.US_ASCII)
        assertTrue(payloadStr.endsWith(" fastdeck@android"))
        val b64 = payloadStr.substringBefore(" ")
        val decoded = Base64.decode(b64, Base64.NO_WRAP)
        assertEquals(524, decoded.size)

        val bb = ByteBuffer.wrap(decoded).order(ByteOrder.LITTLE_ENDIAN)
        assertEquals(64, bb.int)
        val n0inv = bb.int.toLong() and 0xFFFFFFFFL

        val nBytesLE = ByteArray(256).also { bb.get(it) }
        val rrBytesLE = ByteArray(256).also { bb.get(it) }
        val exponent = bb.int
        assertEquals(pub.publicExponent.toInt(), exponent)

        val nFromStruct = BigInteger(1, nBytesLE.reversedArray())
        val rrFromStruct = BigInteger(1, rrBytesLE.reversedArray())
        assertEquals(pub.modulus, nFromStruct)
        val expectedRr = BigInteger.ONE.shiftLeft(4096).mod(pub.modulus)
        assertEquals(expectedRr, rrFromStruct)

        val r32 = BigInteger.ONE.shiftLeft(32)
        val n0 = pub.modulus.mod(r32).toLong()
        val prodMod = (BigInteger.valueOf(n0).multiply(BigInteger.valueOf(n0inv))).mod(r32)
        assertEquals(r32.subtract(BigInteger.ONE), prodMod)

        val token = ByteArray(20) { (it + 7).toByte() }
        val sig = UsbPortController.signAdbTokenPayload(kp.private, token)
        assertEquals(256, sig.size)

        val cipher = Cipher.getInstance("RSA/ECB/PKCS1Padding")
        cipher.init(Cipher.DECRYPT_MODE, pub)
        val recovered = cipher.doFinal(sig)
        val expectedDigestInfo = byteArrayOf(
            0x30, 0x21, 0x30, 0x09, 0x06, 0x05, 0x2B, 0x0E,
            0x03, 0x02, 0x1A, 0x05, 0x00, 0x04, 0x14
        ) + token
        assertArrayEquals(expectedDigestInfo, recovered)
    }

    @Test
    fun `verify sparse chunk planner splits oversized sparse and raw images with valid headers`() {
        val tempSparse = java.io.File.createTempFile("test_sparse", ".img")
        val tempRaw = java.io.File.createTempFile("test_raw", ".img")
        try {
            val blkSz = 4096
            // Build synthetic Sparse image: 3 RAW chunks (2 blocks each = 8192B) + 1 FILL chunk (4 blocks)
            val chunkPayload = ByteArray(blkSz * 2) { 0x5A }
            val sparseBytes = ByteBuffer.allocate(28 + 3 * (12 + chunkPayload.size) + 16)
                .order(ByteOrder.LITTLE_ENDIAN).apply {
                    putInt(com.example.data.SparseChunkPlanner.SPARSE_HEADER_MAGIC)
                    putShort(1); putShort(0); putShort(28); putShort(12)
                    putInt(blkSz); putInt(10); putInt(4); putInt(0)

                    repeat(3) {
                        putShort(com.example.data.SparseChunkPlanner.CHUNK_TYPE_RAW)
                        putShort(0); putInt(2); putInt(12 + chunkPayload.size)
                        put(chunkPayload)
                    }
                    putShort(com.example.data.SparseChunkPlanner.CHUNK_TYPE_FILL)
                    putShort(0); putInt(4); putInt(16)
                    putInt(0x12345678)
                }.array()
            tempSparse.writeBytes(sparseBytes)

            java.io.RandomAccessFile(tempSparse, "r").channel.use { ch ->
                val packets = com.example.data.SparseChunkPlanner.planSparseSplits(
                    channel = ch,
                    baseOffset = 0L,
                    streamSize = sparseBytes.size.toLong(),
                    maxPacketBytes = 10000L // forces splitting across multiple packets!
                )
                assertTrue(packets.size >= 3)
                assertEquals(0L, packets.first().leadingDontCareBlocks)
                assertTrue(packets[1].leadingDontCareBlocks > 0L)
                packets.forEach { pkt ->
                    assertTrue(pkt.totalTransferBytes <= 10000L)
                    val hdr = ByteBuffer.wrap(pkt.buildPacketPrefixHeader()).order(ByteOrder.LITTLE_ENDIAN)
                    assertEquals(com.example.data.SparseChunkPlanner.SPARSE_HEADER_MAGIC, hdr.int)
                }
            }

            // Build synthetic non-4K-aligned Raw image (25,000 bytes) & split across 10 KB chunks
            val rawBytes = ByteArray(25000) { (it and 0xFF).toByte() }
            tempRaw.writeBytes(rawBytes)
            val rawPackets = com.example.data.SparseChunkPlanner.planRawToSparseSplits(
                baseOffset = 0L,
                streamSize = rawBytes.size.toLong(),
                maxPacketBytes = 10000L,
                blockSize = 4096
            )
            assertTrue(rawPackets.size >= 3)
            val coveredReadBytes = rawPackets.flatMap { it.slices }
                .filterIsInstance<com.example.data.SparseChunkPlanner.ChunkSlice.RawSlice>()
                .sumOf { it.readByteLength }
            assertEquals(25000L, coveredReadBytes)
        } finally {
            tempSparse.delete()
            tempRaw.delete()
        }
    }
